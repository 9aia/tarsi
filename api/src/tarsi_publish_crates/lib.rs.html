1//! tarsi-publish-crates — crates.io publisher. Runs
2//! `cargo publish --manifest-path <unit>/Cargo.toml` per unit;
3//! the *caller* supplies units already in dependency order
4//! (plan_release topo-sorts). Retries crates.io 429/rate-limit
5//! responses with exponential backoff + jitter, honoring a
6//! server `Retry-After` hint when cargo surfaces one; each
7//! attempt is bounded by a wall-clock timeout (verify builds on
8//! heavy crates can run for minutes — I-003). "Already exists"
9//! counts as success, verified against the sparse index when
10//! possible (idempotent re-runs). A "failed to select a version"
11//! error naming a release sibling retries too — crates.io's
12//! index lags each upload by seconds–minutes, so dependents
13//! would otherwise fail on a dep published moments earlier.
14

27/// The env var cargo reads for the default-registry token — a
28/// resolved token is always injected under this name regardless
29/// of which `token_env` fallback supplied it.
30pub const CARGO_TOKEN_ENV: &str = "CARGO_REGISTRY_TOKEN";

32/// Default per-attempt wall-clock cap for `cargo publish` —
33/// covers the package verify build plus upload on heavy crates
34/// (tamias-serve/tamias-tui exceeded shorter limits, I-003).
35pub const DEFAULT_TIMEOUT: Duration = Duration::from_secs(15 * 60);
36/// Default base delay for 429 retries — doubles per attempt up
37/// to [`DEFAULT_RETRY_MAX`]. crates.io asks bulk publishers to
38/// slow down by minutes, not seconds.
39pub const DEFAULT_RETRY_BASE: Duration = Duration::from_secs(30);
40/// Default ceiling for computed 429 backoff. A server
41/// `Retry-After` hint is honored verbatim and may exceed this.
42pub const DEFAULT_RETRY_MAX: Duration = Duration::from_secs(5 * 60);

44/// Sparse-index probe signature — `(crate, version) -> listed?`.
45pub type IndexProbe = Arc<dyn Fn(&str, &str) -> bool + Send + Sync>;

47/// crates.io publisher. `repo_root` anchors relative manifest dirs.
48#[derive(Clone)]
49pub struct CratesPublisher {
50    pub repo_root: PathBuf,
51    /// Max attempts per unit when retry_429 is on.
52    pub max_attempts: u32,
53    pub retry_429: bool,
54    /// Per-attempt wall-clock cap for one `cargo publish`
55    /// invocation (verify build + upload). `Duration::ZERO`
56    /// disables the cap.
57    pub timeout: Duration,
58    /// Base delay between 429 retries — doubles each attempt.
59    pub retry_base: Duration,
60    /// Ceiling for computed 429 backoff delays.
61    pub retry_max: Duration,
62    /// Env var NAME consulted for the token when `token_ref` is
63    /// unset or misses (default `CARGO_REGISTRY_TOKEN`).
64    pub token_env: String,
65    /// Optional `cuora://<scope>/<name>` secret ref — takes
66    /// precedence over `token_env`.
67    pub token_ref: Option<String>,
68    /// Unit name → version for every unit in this release. When
69    /// `cargo publish` fails with "failed to select a version"
70    /// naming one of these, the dep was uploaded moments ago and
71    /// crates.io's sparse index is lagging — retry with backoff
72    /// instead of failing the unit (tamias 0.3.0 needed ~4 manual
73    /// re-run rounds without this). Empty = no sibling context,
74    /// index-lag errors fail fast.
75    pub release_versions: BTreeMap<String, String>,
76    /// `(crate, version) -> true` when the sparse index lists the
77    /// version — the "is it actually live" probe behind the
78    /// already-published fast path. Injectable for tests; the
79    /// default GETs `index.crates.io`. Probe errors read as
80    /// "unconfirmed" — the path never blocks on it.
81    pub index_probe: IndexProbe,
82    /// Token resolver — defaults to [`CuoraTokenResolver`]; tests
83    /// inject a fake.
84    pub resolver: Arc<dyn TokenResolver>,
85}

106/// crates.io rate-limit / 429 detection. Scans the *full*
107/// stderr — cargo may print lock-wait noise ahead of the real
108/// error, and the observed bulk-release throttle message is
109/// "too many updates to existing crates" (I-003), which matches
110/// neither "429" nor "too many requests".
111fn is_rate_limited(stderr: &str) -> bool {
112    let s = stderr.to_lowercase();
113    s.contains("429")
114        || s.contains("rate limit")
115        || s.contains("rate-limit")
116        || s.contains("rate_limit")
117        || s.contains("too many")
118}
119
120fn is_already_published(stderr: &str) -> bool {
121    stderr.contains("already exists")
122        || stderr.contains("is already uploaded")
123        || stderr.contains("previously uploaded")
124}

126/// crates.io sparse-index URL for `name`. Prefix layout: len 1 →
127/// `1/`, 2 → `2/`, 3 → `3/<c0>/`, ≥4 → `<c0c1>/<c2c3>/` — all
128/// lowercase (crate names are case-insensitive on the index).
129fn index_url(name: &str) -> String {
130    let n = name.to_ascii_lowercase();
131    let path = match n.len() {
132        1 => format!("1/{n}"),
133        2 => format!("2/{n}"),
134        3 => format!("3/{}/{n}", &n[..1]),
135        _ => format!("{}/{}/{n}", &n[..2], &n[2..4]),
136    };
137    format!("https://index.crates.io/{path}")
138}

140/// Default [`CratesPublisher::index_probe`]: GET the sparse index
141/// entry and look for `"vers":"<version>"` — one JSON object per
142/// line, versions as a string field. Any transport/HTTP error
143/// reads as unconfirmed (the already-published path reports the
144/// caveat in `detail` rather than failing the unit).
145fn default_index_probe(name: &str, version: &str) -> bool {
146    let Ok(resp) = ureq::get(&index_url(name))
147        .header("User-Agent", "tarsi-publish-crates")
148        .call()
149    else {
150        return false;
151    };
152    let Ok(body) = resp.into_body().read_to_string() else {
153        return false;
154    };
155    body.contains(&format!("\"vers\":\"{version}\""))
156}

158/// crates.io sparse-index lag signature: cargo's resolver reports
159/// `failed to select a version for the requirement `NAME = "…"``
160/// for a dep version uploaded seconds ago (the dep's own publish
161/// succeeded — only the index trails). Returns true only when the
162/// missing crate is a release sibling: a foreign missing dep is a
163/// real resolution failure and must not burn retry budget.
164fn is_index_lag_for_sibling(stderr: &str, siblings: &BTreeMap<String, String>) -> bool {
165    let Some(pos) = stderr.find("failed to select a version") else {
166        return false;
167    };
168    let rest = &stderr[pos..];
169    let Some(open) = rest.find('`') else {
170        return false;
171    };
172    let req = &rest[open + 1..];
173    let Some(close) = req.find('`') else {
174        return false;
175    };
176    let name = req[..close].split('=').next().unwrap_or("").trim();
177    !name.is_empty() && siblings.contains_key(name)
178}

180/// Server-directed retry hint. cargo doesn't always surface the
181/// `Retry-After` header, but when it does (or when crates.io's
182/// message says "try again in N seconds") we honor it verbatim —
183/// it beats any computed backoff.
184fn parse_retry_after(stderr: &str) -> Option<u64> {
185    for line in stderr.lines() {
186        let l = line.to_lowercase();
187        if !(l.contains("retry") || l.contains("try again")) {
188            continue;
189        }
190        // Last integer on the line is the delay in seconds
191        // ("Retry-After: 120", "try again in 60 seconds").
192        let mut last = None;
193        let mut cur = String::new();
194        for c in l.chars() {
195            if c.is_ascii_digit() {
196                cur.push(c);
197            } else {
198                if !cur.is_empty() {
199                    last = cur.parse::<u64>().ok().or(last);
200                    cur.clear();
201                }
202            }
203        }
204        if !cur.is_empty() {
205            last = cur.parse::<u64>().ok().or(last);
206        }
207        if let Some(n) = last {
208            if n > 0 {
209                return Some(n);
210            }
211        }
212    }
213    None
214}

216/// Exponential backoff with jitter: `base * 2^(attempt-1)` capped
217/// at `max`, plus up to 25% jitter so parallel publishers don't
218/// retry in lockstep. A `Retry-After` hint in `stderr` wins.
219fn retry_delay(attempt: u32, stderr: &str, base: Duration, max: Duration) -> Duration {
220    if let Some(secs) = parse_retry_after(stderr) {
221        return Duration::from_secs(secs);
222    }
223    let capped = base
224        .saturating_mul(1u32 << attempt.saturating_sub(1).min(20))
225        .min(max);
226    // Jitter: up to +25% of the capped delay, from wall-clock
227    // nanos (no rng dep for a release tool).
228    let span = capped.as_nanos() / 4 + 1;
229    let nanos = SystemTime::now()
230        .duration_since(UNIX_EPOCH)
231        .map(|d| d.subsec_nanos() as u128)
232        .unwrap_or(0);
233    capped + Duration::from_nanos((nanos % span) as u64)
234}

236/// One finished `cargo publish` invocation — exit status plus
237/// captured output (stderr is what carries the diagnostics).
238struct Captured {
239    /// `None` when the attempt was killed on timeout.
240    status: Option<ExitStatus>,
241    stderr: String,
242}
243
244/// How one attempt ended.
245enum Attempt {
246    Exited(Captured),
247    TimedOut(Captured),
248}
249
250/// Spawn `cmd` with output redirected to temp files, poll for
251/// exit, and enforce `timeout` (`Duration::ZERO` = no cap).
252///
253/// Files — not pipes — carry the output: a killed `cargo` can
254/// leave grandchildren (rustc, the registry client) holding pipe
255/// write-ends open, which would block a `read_to_end` join long
256/// after the timeout fired (observed: a 500ms cap still waited
257/// out the child's full 30s sleep). File reads return whatever
258/// was written and never block on a still-open fd.
259fn run_attempt(mut cmd: Command, timeout: Duration) -> Result<Attempt, PublishError> {
260    static SEQ: AtomicU64 = AtomicU64::new(0);
261    let seq = SEQ.fetch_add(1, Ordering::Relaxed);
262    let base = std::env::temp_dir().join(format!("tarsi-publish-{}-{seq}", std::process::id()));
263    let out_path = base.with_extension("out");
264    let err_path = base.with_extension("err");
265    cmd.stdout(Stdio::from(std::fs::File::create(&out_path)?))
266        .stderr(Stdio::from(std::fs::File::create(&err_path)?));
267    let mut child = cmd.spawn()?;
268
269    let deadline = (!timeout.is_zero()).then(|| Instant::now() + timeout);
270    let status = loop {
271        if let Some(status) = child.try_wait()? {
272            break Some(status);
273        }
274        if deadline.is_some_and(|d| Instant::now() >= d) {
275            let _ = child.kill();
276            let _ = child.wait();
277            break None;
278        }
279        std::thread::sleep(Duration::from_millis(100));
280    };
281    // cargo diagnostics live on stderr; stdout is drained-and-
282    // dropped (a verify build's compile log is noise here).
283    let stderr_bytes = std::fs::read(&err_path).unwrap_or_default();
284    let _ = std::fs::remove_file(&out_path);
285    let _ = std::fs::remove_file(&err_path);
286    let captured = Captured {
287        status,
288        stderr: String::from_utf8_lossy(&stderr_bytes).into_owned(),
289    };
290    Ok(match status {
291        Some(_) => Attempt::Exited(captured),
292        None => Attempt::TimedOut(captured),
293    })
294}

296/// Tail of stderr for error details — lock-wait noise leads,
297/// the real error trails (I-003 truncated details hid it).
298fn stderr_tail(stderr: &str, lines: usize) -> String {
299    let all: Vec<&str> = stderr.lines().collect();
300    all.iter()
301        .rev()
302        .take(lines)
303        .rev()
304        .copied()
305        .collect::<Vec<_>>()
306        .join(" | ")
307}

309impl Publisher for CratesPublisher {
310    fn target(&self) -> PublishTarget {
311        PublishTarget::CratesIo
312    }
313
314    fn publish(&self, job: &PublishJob) -> Result<PublishReport, PublishError> {
315        let manifest = self.repo_root.join(&job.manifest_dir).join("Cargo.toml");
316        if job.dry_run {
317            return Ok(PublishReport {
318                unit: job.unit.clone(),
319                target: self.target(),
320                ok: true,
321                detail: format!("dry-run: would cargo publish {}", manifest.display()),
322                marker: None,
323            });
324        }
325        // Resolve the token once per unit — after the dry-run
326        // early-return so dry-runs never touch cuora/env secrets.
327        let token = self
328            .resolver
329            .resolve(&self.token_env, self.token_ref.as_deref())?;
330        let attempts = if self.retry_429 { self.max_attempts } else { 1 };
331        let mut last_err = String::new();
332        for attempt in 1..=attempts {
333            let mut cmd = Command::new("cargo");
334            cmd.arg("publish").arg("--manifest-path").arg(&manifest);
335            // Inject under the name cargo actually reads — the
336            // configured `token_env` only names the fallback.
337            if let Some(t) = &token {
338                cmd.env(CARGO_TOKEN_ENV, t);
339            }
340            match run_attempt(cmd, self.timeout)? {
341                Attempt::TimedOut(c) => {
342                    return Err(PublishError::Msg(format!(
343                        "cargo publish {} timed out after {:?}: {}",
344                        job.unit,
345                        self.timeout,
346                        stderr_tail(&c.stderr, 8)
347                    )));
348                }
349                Attempt::Exited(c) if c.status.is_some_and(|s| s.success()) => {
350                    return Ok(PublishReport {
351                        unit: job.unit.clone(),
352                        target: self.target(),
353                        ok: true,
354                        detail: format!("published {} (attempt {attempt})", job.version),
355                        marker: None,
356                    });
357                }
358                Attempt::Exited(c) => {
359                    if is_already_published(&c.stderr) {
360                        // Idempotent re-run: the version is live.
361                        // Confirm against the sparse index so the
362                        // report says *verified*, not just "cargo
363                        // said so" — an unconfirmed probe (index
364                        // lag right after the first upload, or a
365                        // network blip) downgrades the detail but
366                        // never fails the unit: crates.io's API
367                        // already asserted the version exists.
368                        let verified = (self.index_probe)(&job.unit, &job.version);
369                        return Ok(PublishReport {
370                            unit: job.unit.clone(),
371                            target: self.target(),
372                            ok: true,
373                            detail: if verified {
374                                "already published (index-verified)".to_string()
375                            } else {
376                                "already published (index did not confirm — propagation lag or probe failure)".to_string()
377                            },
378                            marker: Some("already_published".to_string()),
379                        });
380                    }
381                    let rate_limited = is_rate_limited(&c.stderr);
382                    let index_lag = is_index_lag_for_sibling(&c.stderr, &self.release_versions);
383                    last_err = stderr_tail(&c.stderr, 8);
384                    let retryable = self.retry_429 && (rate_limited || index_lag);
385                    if !retryable || attempt == attempts {
386                        break;
387                    }
388                    let delay = retry_delay(attempt, &c.stderr, self.retry_base, self.retry_max);
389                    std::thread::sleep(delay);
390                }
391            }
392        }
393        Err(PublishError::Msg(format!(
394            "cargo publish {} failed: {last_err}",
395            job.unit
396        )))
397    }
398}

433    fn backoff_grows_and_caps() {
434        let base = Duration::from_secs(10);
435        let max = Duration::from_secs(60);
436        let d1 = retry_delay(1, "", base, max);
437        let d3 = retry_delay(3, "", base, max);
438        // attempt1: 10s (+jitter ≤2.5s); attempt3: 40s (+jitter ≤10s)
439        assert!(d1 >= base && d1 < base + Duration::from_secs(3));

441        // cap
442        let d9 = retry_delay(9, "", base, max);
443        assert!(d9 <= max + max / 4);
444        // Retry-After wins over computation
445        let hinted = retry_delay(1, "Retry-After: 300", base, max);
446        assert_eq!(hinted, Duration::from_secs(300));

487        S: FnOnce(&std::path::Path) -> String,
488        F: FnOnce(&PathBuf),

498        }

500        std::env::set_var("PATH", format!("{}:{old_path}", dir.path().display()));
501        f(&dir.path().to_path_buf());
502        std::env::set_var("PATH", old_path);
503    }

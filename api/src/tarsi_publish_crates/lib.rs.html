1//! tarsi-publish-crates — crates.io publisher. Runs
2//! `cargo publish --manifest-path <unit>/Cargo.toml` per unit;
3//! the *caller* supplies units already in dependency order
4//! (plan_release topo-sorts). Retries on crates.io 429/rate-limit
5//! output with linear backoff; "already exists" counts as success
6//! (idempotent re-runs — release.sh semantics).
7
8use std::path::PathBuf;
9use std::process::Command;
10use std::sync::Arc;
11use std::time::Duration;
12
13use tarsi_core::PublishTarget;
14use tarsi_publish::{
15    CuoraTokenResolver, PublishError, PublishJob, PublishReport, Publisher, TokenResolver,
16};

18/// The env var cargo reads for the default-registry token — a
19/// resolved token is always injected under this name regardless
20/// of which `token_env` fallback supplied it.
21pub const CARGO_TOKEN_ENV: &str = "CARGO_REGISTRY_TOKEN";
22
23/// crates.io publisher. `repo_root` anchors relative manifest dirs.
24#[derive(Clone)]
25pub struct CratesPublisher {
26    pub repo_root: PathBuf,
27    /// Max attempts per unit when retry_429 is on.
28    pub max_attempts: u32,
29    pub retry_429: bool,
30    /// Env var NAME consulted for the token when `token_ref` is
31    /// unset or misses (default `CARGO_REGISTRY_TOKEN`).
32    pub token_env: String,
33    /// Optional `cuora://<scope>/<name>` secret ref — takes
34    /// precedence over `token_env`.
35    pub token_ref: Option<String>,
36    /// Token resolver — defaults to [`CuoraTokenResolver`]; tests
37    /// inject a fake.
38    pub resolver: Arc<dyn TokenResolver>,
39}
40
41impl CratesPublisher {
42    #[must_use]
43    pub fn new(repo_root: PathBuf) -> Self {
44        Self {
45            repo_root,
46            max_attempts: 5,
47            retry_429: true,
48            token_env: CARGO_TOKEN_ENV.to_string(),
49            token_ref: None,
50            resolver: Arc::new(CuoraTokenResolver::new()),
51        }
52    }
53}
54
55fn is_rate_limited(stderr: &str) -> bool {
56    stderr.contains("429") || stderr.contains("rate limit") || stderr.contains("too many requests")
57}
58
59fn is_already_published(stderr: &str) -> bool {
60    stderr.contains("already exists")
61        || stderr.contains("is already uploaded")
62        || stderr.contains("previously uploaded")
63}
64
65impl Publisher for CratesPublisher {
66    fn target(&self) -> PublishTarget {
67        PublishTarget::CratesIo
68    }
69
70    fn publish(&self, job: &PublishJob) -> Result<PublishReport, PublishError> {
71        let manifest = self.repo_root.join(&job.manifest_dir).join("Cargo.toml");
72        if job.dry_run {
73            return Ok(PublishReport {
74                unit: job.unit.clone(),
75                target: self.target(),
76                ok: true,
77                detail: format!("dry-run: would cargo publish {}", manifest.display()),
78            });
79        }
80        // Resolve the token once per unit — after the dry-run
81        // early-return so dry-runs never touch cuora/env secrets.
82        let token = self
83            .resolver
84            .resolve(&self.token_env, self.token_ref.as_deref())?;
85        let attempts = if self.retry_429 { self.max_attempts } else { 1 };
86        let mut last_err = String::new();
87        for attempt in 1..=attempts {
88            let mut cmd = Command::new("cargo");
89            cmd.arg("publish").arg("--manifest-path").arg(&manifest);
90            // Inject under the name cargo actually reads — the
91            // configured `token_env` only names the fallback.
92            if let Some(t) = &token {
93                cmd.env(CARGO_TOKEN_ENV, t);
94            }
95            let out = cmd.output()?;
96            let stderr = String::from_utf8_lossy(&out.stderr).into_owned();
97            if out.status.success() {
98                return Ok(PublishReport {
99                    unit: job.unit.clone(),
100                    target: self.target(),
101                    ok: true,
102                    detail: format!("published {} (attempt {attempt})", job.version),
103                });
104            }
105            if is_already_published(&stderr) {
106                return Ok(PublishReport {
107                    unit: job.unit.clone(),
108                    target: self.target(),
109                    ok: true,
110                    detail: "already published".to_string(),
111                });
112            }
113            last_err = stderr.lines().take(3).collect::<Vec<_>>().join(" | ");
114            if !(self.retry_429 && is_rate_limited(&last_err)) || attempt == attempts {
115                break;
116            }
117            std::thread::sleep(Duration::from_secs(u64::from(attempt) * 15));
118        }
119        Err(PublishError::Msg(format!(
120            "cargo publish {} failed: {last_err}",
121            job.unit
122        )))
123    }
124}

131    #[test]
132    fn rate_limit_detection() {
133        assert!(is_rate_limited("error: 429 Too Many Requests"));
134        assert!(is_rate_limited("you have hit your rate limit"));
135        assert!(!is_rate_limited("error: no such package"));
136        assert!(is_already_published(
137            "error: crate version 1.0.0 is already uploaded"
138        ));
139    }

141    #[test]
142    fn dry_run_never_spawns() {
143        let p = CratesPublisher::new(PathBuf::from("/nonexistent"));
144        let job = PublishJob {
145            unit: "u".into(),
146            version: "1.0.0".into(),
147            manifest_dir: "crates/u".into(),
148            tag: "u-v1.0.0".into(),
149            notes: String::new(),
150            assets: Vec::new(),
151            commitish: None,
152            dry_run: true,
153        };
154        let r = p.publish(&job).unwrap();
155        assert!(r.ok && r.detail.contains("dry-run"));
156    }

171    /// Resolver seam — fixed token or error, env ignored.
172    struct FakeResolver(Result<Option<String>, String>);
173
174    impl TokenResolver for FakeResolver {
175        fn resolve(
176            &self,
177            _token_env: &str,
178            _token_ref: Option<&str>,
179        ) -> Result<Option<String>, PublishError> {
180            match &self.0 {
181                Ok(v) => Ok(v.clone()),
182                Err(e) => Err(PublishError::Msg(e.clone())),
183            }
184        }
185    }

187    #[test]
188    fn resolved_token_is_injected_as_cargo_token_env() {
189        use std::os::unix::fs::PermissionsExt;
190        let dir = tempfile::tempdir().unwrap();
191        // Fake `cargo`: records CARGO_REGISTRY_TOKEN, exits 0.
192        let capture = dir.path().join("token.txt");
193        let fake = dir.path().join("cargo");
194        std::fs::write(
195            &fake,
196            format!(
197                "#!/bin/sh\nprintf '%s' \"${{CARGO_REGISTRY_TOKEN:-}}\" > '{}'\nexit 0\n",
198                capture.display()
199            ),
200        )
201        .unwrap();
202        std::fs::set_permissions(&fake, std::fs::Permissions::from_mode(0o755)).unwrap();
203        // Prepend the stub dir to PATH for this test process.
204        let old_path = std::env::var("PATH").unwrap_or_default();
205        std::env::set_var("PATH", format!("{}:{old_path}", dir.path().display()));
206
207        let mut p = CratesPublisher::new(dir.path().to_path_buf());
208        p.retry_429 = false;
209        p.resolver = Arc::new(FakeResolver(Ok(Some("sekret-token".into()))));
210        p.token_ref = Some("cuora://tamias/crates-token".into());
211        // manifest_dir is repo-root-relative; point repo_root at
212        // the tempdir so the (nonexistent) manifest path is fine —
213        // the fake cargo never reads it.
214        let r = p.publish(&job(false));
215        std::env::set_var("PATH", old_path);
216        let r = r.unwrap();
217        assert!(r.ok, "{}", r.detail);
218        assert_eq!(std::fs::read_to_string(&capture).unwrap(), "sekret-token");
219    }

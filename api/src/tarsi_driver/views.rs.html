1//! views — the driver's manifest: every ViewSpec plus the query
2//! resolvers that feed them (the pavo read-only surface, PL-014
3//! D3 remainder).
4//!
5//! Read path: every query is one `tarsi-serve` one-shot op — see
6//! `data.rs`. `plan`/`changelog` take `dir`/`profile`/`range`
7//! params (`--param dir=<repo>`, `--param profile=9aia-release`,
8//! `--param range=HEAD~3..HEAD` on the pavo wire); absent `dir`
9//! means `.`. A failed serve call degrades to an `{"error": …}`
10//! payload inside an `ok:true` response, matching the sibling
11//! drivers' contract — the views stay renderable with no
12//! `tarsi-serve` on PATH.
13//!
14//! Surface mapping:
15//!
16//! | pavo view       | tarsi-serve op | data                              |
17//! |-----------------|----------------|-----------------------------------|
18//! | `tarsi:status`  | `manifest`     | binary, version, ops, default dir |
19//! | `tarsi:profiles`| `profiles`     | release-profile list (all roots)  |
20//! | `tarsi:plan`    | `plan`         | release-plan digest + unit rows   |
21//! | `tarsi:changelog`| `changelog`   | per-unit changelog markdown      |
22//!
23//! `tarsi:plan` answers "what would a release of dir X do?" —
24//! bumps, tags, publish-target switches, unattributed commits,
25//! ordered steps — without mutating anything (`plan` is read-only
26//! in the serve; `release` is never called).
27

41fn col(title: &str, children: Vec<ViewNode>) -> ViewNode {
42    kit::column(kit::base().title(title), children)
43}
44
45fn kv(title: &str, query: &str) -> ViewNode {
46    kit::kv(kit::base().title(title).query(query), Vec::new())
47}
48
49fn md(title: &str, text: &str) -> ViewNode {
50    kit::markdown(kit::base().title(title), text)
51}
52
53fn table(title: &str, query: &str, columns: &[(&str, &str)]) -> ViewNode {
54    kit::sortable_table(
55        kit::base().title(title).query(query),
56        columns
57            .iter()
58            .map(|(key, t)| kit::table_column(key, t))
59            .collect::<Vec<TableColumn>>(),
60        Vec::new(),
61    )
62}

64fn spec(
65    id: &str,
66    title: &str,
67    route: &str,
68    description: &str,
69    queries: &[&str],
70    root: ViewNode,
71) -> ViewSpec {
72    let mut view = kit::view(id, title, route, root).description(description);
73    for q in queries {
74        view = view.query(kit::binding(q, "query"));
75    }
76    view.build()
77}

148/// The `manifest` op result — built with `pavo_driver_kit`'s
149/// `Manifest` fluent builder.
150pub fn manifest() -> DriverManifest {
151    let mut navbar = kit::contribution("plan", "Release plan", "/tarsi/plan");
152    navbar.priority = 20;
153    let mut dashboard = kit::contribution("status", "Tarsi status", "/tarsi/status");
154    dashboard.priority = 30;
155
156    let mut manifest = kit::Manifest::new("tarsi", "Tarsi")
157        .version(env!("CARGO_PKG_VERSION"))
158        .navbar(navbar)
159        .dashboard(dashboard);
160    for view in specs() {
161        manifest = manifest.view(view);
162    }
163    manifest.build()
164}

168/// Every query name the driver answers — the `query` op rejects
169/// anything outside this set with `ok:false`.
170pub fn known_query(name: &str) -> bool {
171    matches!(
172        name,
173        "summary" | "profiles" | "plan" | "units" | "changelog"
174    )
175}

177/// `{"op":"query","query":name,"params":{…}}` → the data payload.
178/// Failures degrade to `{"error": …}` objects, never `ok:false`.
179pub fn query_data(name: &str, params: &Map<String, Value>, env: Env<'_>) -> Value {
180    match name {
181        "summary" => summary(params, env),
182        "profiles" => profiles_rows(env),
183        "plan" => plan_digest(params, env),
184        "units" => units_rows(params, env),
185        "changelog" => changelog_data(params, env),
186        other => json!({"error": format!("unknown query '{other}'")}),
187    }
188}
189
190/// `summary` query — the status view's kv object: which
191/// `tarsi-serve` binary answered (or why none did), its version +
192/// ops, and the dir plan queries default to.
193fn summary(params: &Map<String, Value>, env: Env<'_>) -> Value {
194    let dir = data::repo_dir(params);
195    let mut out = Map::new();
196    out.insert(
197        "binary".into(),
198        json!(env("TARSI_SERVE")
199            .filter(|s| !s.is_empty())
200            .unwrap_or_else(|| "tarsi-serve".to_string())),
201    );
202    out.insert("dir".into(), json!(dir));
203    match data::manifest_payload(env) {
204        Ok(v) => {
205            out.insert("serve".into(), json!("ok"));
206            out.insert("version".into(), v["version"].clone());
207            out.insert("ops".into(), v["ops"].clone());
208        }
209        Err(e) => {
210            out.insert("serve".into(), json!("error"));
211            out.insert("error".into(), json!(e));
212        }
213    }
214    Value::Object(out)
215}

217/// `profiles` query — `tarsi-serve profiles` rows verbatim
218/// (`{profiles: […]}` unwraps to the array the table binds).
219fn profiles_rows(env: Env<'_>) -> Value {
220    match data::profiles_payload(env) {
221        Ok(v) => v["profiles"].clone(),
222        Err(e) => json!({"error": e}),
223    }
224}

226/// The `plan` op payload → `{specs, plan}`; errors degrade.
227fn plan_or_error(params: &Map<String, Value>, env: Env<'_>) -> Value {
228    match data::plan_payload(params, env) {
229        Ok(v) => v,
230        Err(e) => json!({"error": e}),
231    }
232}

234/// `plan` query — the digest half of the plan view: target dir,
235/// the effective profile/range params, ordered steps, and counts.
236fn plan_digest(params: &Map<String, Value>, env: Env<'_>) -> Value {
237    let payload = plan_or_error(params, env);
238    if let Some(e) = payload.get("error").and_then(Value::as_str) {
239        return json!({"error": e});
240    }
241    let plan = &payload["plan"];
242    let units = plan["units"].as_array().map(Vec::len).unwrap_or(0);
243    let releasing = plan["units"]
244        .as_array()
245        .map(|us| {
246            us.iter()
247                .filter(|u| u["releases"].as_bool().unwrap_or(false))
248                .count()
249        })
250        .unwrap_or(0);
251    json!({
252        "dir": data::repo_dir(params),
253        "profile": params.get("profile").cloned().unwrap_or(Value::Null),
254        "range": params.get("range").cloned().unwrap_or(Value::Null),
255        "steps": plan["steps"],
256        "units": units,
257        "releasing": releasing,
258        "unattributed": plan["unattributed"].as_array().map(Vec::len).unwrap_or(0),
259    })
260}

262/// `spec.publish` targets → `"crates.io:on, ghreleases:off"` —
263/// the effective switch (`publish.<target>.enabled`), not just
264/// the declaration, since the serve already resolved it.
265fn targets_cell(u: &Value) -> String {
266    u["publish_targets"]
267        .as_array()
268        .map(|ts| {
269            ts.iter()
270                .map(|t| {
271                    format!(
272                        "{}:{}",
273                        t["target"].as_str().unwrap_or("?"),
274                        if t["enabled"].as_bool().unwrap_or(false) {
275                            "on"
276                        } else {
277                            "off"
278                        }
279                    )
280                })
281                .collect::<Vec<_>>()
282                .join(", ")
283        })
284        .unwrap_or_default()
285}

287/// One plan unit → table row.
288fn unit_row(u: &Value) -> Value {
289    json!({
290        "unit": u["spec"]["name"],
291        "path": u["spec"]["path"],
292        "bump": u["bump"],
293        "new_version": u["new_version"],
294        "tag": u["tag"],
295        "releases": u["releases"],
296        "commits": u["commits"],
297        "targets": targets_cell(u),
298    })
299}

301/// `units` query — per-unit release rows for the plan table.
302fn units_rows(params: &Map<String, Value>, env: Env<'_>) -> Value {
303    let payload = plan_or_error(params, env);
304    if let Some(e) = payload.get("error").and_then(Value::as_str) {
305        return json!({"error": e});
306    }
307    Value::Array(
308        payload["plan"]["units"]
309            .as_array()
310            .map(|us| us.iter().map(unit_row).collect())
311            .unwrap_or_default(),
312    )
313}

324// ---------------------------------------------------------------- view op
325
326/// `{"op":"view","view":id,"params":{…}}` → [`ViewResult`].
327///
328/// Every view resolves its bound queries against `params`; the
329/// `changelog` view additionally inlines the resolved markdown
330/// into its root (markdown nodes aren't query-bound — the spec is
331/// personalized per request, the myco-driver pattern).
332pub fn view_result(
333    view_id: &str,
334    params: &Map<String, Value>,
335    env: Env<'_>,
336) -> Result<ViewResult, String> {
337    let spec = specs()
338        .into_iter()
339        .find(|v| v.id == view_id)
340        .ok_or_else(|| format!("unknown view: {view_id}"))?;
341
342    let mut data_map = Map::new();
343    for b in &spec.queries {
344        data_map.insert(b.name.clone(), query_data(&b.name, params, env));
345    }
346
347    let root = if view_id == "changelog" {
348        changelog_root(&data_map["changelog"])
349    } else {
350        spec.root.clone()
351    };
352
353    Ok(ViewResult {
354        view: ViewSpec { root, ..spec },
355        data: data_map,
356    })
357}

359/// The `changelog` view's personalized tree — one markdown node
360/// per releasing unit (`## <unit> <version>` heading already lives
361/// in each entry's markdown; the node title carries unit+tag), or
362/// a single error/note node when the serve call degraded.
363fn changelog_root(payload: &Value) -> ViewNode {
364    if let Some(e) = payload.get("error").and_then(Value::as_str) {
365        return col(
366            "Changelog preview",
367            vec![md("Preview", &format!("*(error: {e})*"))],
368        );
369    }
370    let entries = payload["changelogs"].as_array();
371    let Some(entries) = entries else {
372        return col(
373            "Changelog preview",
374            vec![md("Preview", "*(no changelog payload)*")],
375        );
376    };
377    if entries.is_empty() {
378        return col(
379            "Changelog preview",
380            vec![md(
381                "Preview",
382                "*(nothing to release — no changelog preview)*",
383            )],
384        );
385    }
386    let mut nodes = Vec::new();
387    for (i, entry) in entries.iter().enumerate() {
388        let unit = entry["unit"].as_str().unwrap_or("unit");
389        let version = entry["version"].as_str().unwrap_or("");
390        let title = if version.is_empty() {
391            unit.to_string()
392        } else {
393            format!("{unit} {version}")
394        };
395        // The tag already rides inside each entry's markdown heading.
396        let text = entry["markdown"].as_str().unwrap_or("");
397        let mut node = md(&title, text);
398        // Keep node ids stable + unique across units.
399        if let ViewNode::Markdown(m) = &mut node {
400            m.base.id = Some(format!("changelog-{i}"));
401        }
402        nodes.push(node);
403    }
404    col("Changelog preview", nodes)
405}

419    fn env_of(vars: &[(&str, &str)]) -> impl Fn(&str) -> Option<String> {
420        let map: BTreeMap<String, String> = vars
421            .iter()

423            .collect();
424        move |k| map.get(k).cloned()

1//! tarsi-driver — library half of `tamias-driver-tarsi`, the tamias
2//! content driver for `type: tarsi` release-profile packs (PL-014 D3).
3//!
4//! A `type: tarsi` pack is a **release profile**: a payload whose
5//! root carries `tarsi.{yaml,yml,toml}` (a plain
6//! [`tarsi_core::Config`] document — no `[profile]` table; `profile`
7//! is the *selector* key) plus optional extras (`styleguide.md`,
8//! commit maps, …). See `tarsi-config`'s `profile` module for the
9//! discovery contract this driver installs into.
10//!
11//! # Where install places the profile
12//!
13//! tamias dispatches `install` *before* its store commit, with
14//! `--dir` = the payload root; `uninstall`/`deploy`/`status` get the
15//! committed store entry dir (`<store>/[@scope/]<name>@<label>/`).
16//! tarsi's profile discovery already treats the tamias store as a
17//! first-class source — `store_profiles` finds every
18//! `<store>/name@label/` carrying a `tarsi.*` and `find_dir` picks
19//! the newest version label. **The store commit is therefore the
20//! placement**: `install` validates the payload and reports; the
21//! pack becomes resolvable the moment tamias commits it, versioned
22//! by label.
23//!
24//! The driver deliberately does *not* copy payloads into
25//! `~/.config/tarsi/profiles/<name>/` by default: dir profiles are
26//! searched before the store and win `find_dir`, and they carry no
27//! version — an unversioned copy would shadow every store version,
28//! including ones `tamias remove` already dropped (the host only
29//! dispatches `uninstall` on the *last* same-name sibling, T-029).
30//!
31//! Opt-in escape hatch: `--config` may name a `profilesDir`. Then
32//! `install`/`deploy` also materialize an **unversioned dir-profile
33//! copy** at `<profilesDir>/<name>/` (scoped names nest
34//! `@scope/<name>/`) stamped with a `.tamias-driver-tarsi.json`
35//! marker, and `uninstall` removes marker-bearing copies it finds
36//! under the profiles roots. Use it to pin a profile outside the
37//! evictable store — knowing the copy shadows store resolution.
38//!
39//! # Verbs (the tamias driver contract)
40//!
41//! `<verb> --dir <d> --name <n> [--config <f>]`:
42//!
43//! - `capabilities` — `{"capabilities": ["tamias"], "types":
44//!   ["tarsi"], "verbs": […]}`; tamias's discovery probe.
45//! - `install` — validate the payload is a well-formed profile
46//!   (`tarsi.*` present and parses as `Config`; `tamias.yaml`'s
47//!   `type` must be `tarsi` when present), then optionally write
48//!   the `profilesDir` copy. JSON report; nothing else is applied.
49//! - `deploy` — refresh the operator-facing state: re-validate the
50//!   installed entry and re-sync the `profilesDir` copy when
51//!   configured; reports how `tarsi` resolves the name.
52//! - `uninstall` — remove marker-bearing `profilesDir` copies under
53//!   the profiles roots (`$TARSI_PROFILES_DIR`,
54//!   `~/.config/tarsi/profiles`, plus the configured dir). The store
55//!   entry itself is dropped by tamias after this verb returns.
56//! - `status` — resolution report: which dir `find_dir` picks for
57//!   the name, from which source, and every installed candidate.
58//! - `health` — discovery probe: profiles roots + visible profile
59//!   count. No services exist for a profile pack.
60//! - `logs` / `hooks` — empty reports; `type: tarsi` packs run no
61//!   units and declare no forge hooks.
62//! - `config-schema` — JSON Schema (draft 2020-12) of the optional
63//!   `--config` file (see [`DriverConfig`]).
64//!
65//! # The pavo read-only surface
66//!
67//! The same binary is also a **pavo view driver** (PL-014 D3
68//! remainder): `manifest` prints the [`pavo_protocol::DriverManifest`]
69//! and `serve` answers ndjson JSON-RPC (`manifest`/`views`/`view`/
70//! `query`) over [`views`] — release plans, changelog previews, and
71//! publish-profile state, all read through `tarsi-serve` one-shot
72//! ops ([`data`]). The surfaces are disjoint by argv: pavo's probes
73//! (`manifest`/`capabilities`/`serve`) take no `--dir`/`--name`,
74//! tamias's verbs always carry them — `capabilities` answers a
75//! merged payload satisfying both discoverers. Nothing on the pavo
76//! side mutates: the serve ops called are `manifest`, `profiles`,
77//! `plan`, `changelog` — `release` is never dispatched.
78//!
79//! See `views.rs` for the view/ query mapping and `data.rs` for
80//! the `tarsi-serve` invocation (`$TARSI_SERVE` overrides the
81//! binary, default `tarsi-serve` on PATH).

121/// The optional `--config` document (YAML; JSON parses too):
122///
123/// ```yaml
124/// profilesDir: ~/.config/tarsi/profiles   # opt-in unversioned copy

128/// `install`/`deploy` maintain and `uninstall` removes. Relative

131/// (the default — see the module docs for why).

134    /// Root for the opt-in dir-profile copy.
135    pub profiles_dir: Option<PathBuf>,

138/// `capabilities` — merged for both consumers of this binary.
139/// pavo's discovery checks `capabilities`/`verbs` for
140/// `"pavo"`/`"serve"`; tamias's checks `types` for the content kind
141/// and warns on undeclared `verbs`. One JSON answers both: the
142/// kit's `{capabilities, verbs}` shape plus `"types": ["tarsi"]`,
143/// `"tamias"` in `capabilities`, and every verb folded into
144/// `verbs`.
145pub fn capabilities_value() -> Value {
146    let m = views::manifest();
147    let mut v = pavo_driver_kit::capabilities_value(&m);
148    v["views"] = json!(m.views.len());
149    v["types"] = json!(["tarsi"]);
150    v["capabilities"] = json!(["pavo", "tamias"]);
151    let mut verbs: Vec<Value> = v["verbs"].as_array().cloned().unwrap_or_default();
152    for extra in [
153        "capabilities",
154        "config-schema",
155        "install",
156        "deploy",
157        "uninstall",
158        "status",
159        "health",
160        "logs",
161        "hooks",
162    ] {
163        if !verbs.iter().any(|x| x.as_str() == Some(extra)) {
164            verbs.push(json!(extra));
165        }
166    }
167    v["verbs"] = Value::Array(verbs);
168    v
169}

190/// Load `--config`: `None` → defaults; a path → YAML (JSON is valid
191/// YAML) with `profilesDir` rebased onto the config file's dir.
192///
193/// # Errors
194/// Unreadable file, non-mapping document, unknown keys, or a
195/// non-string `profilesDir`.
196pub fn load_config(path: Option<&Path>, env: Env<'_>) -> Result<DriverConfig, String> {
197    let Some(path) = path else {
198        return Ok(DriverConfig::default());
199    };
200    let text = fs::read_to_string(path)
201        .map_err(|e| format!("{WHO}: cannot read config {}: {e}", path.display()))?;
202    let value: Value = YamlFormat
203        .parse(&text)
204        .map_err(|e| format!("{WHO}: {}: invalid driver config: {e}", path.display()))?;
205    let Some(map) = value.as_object() else {
206        return Err(format!(
207            "{WHO}: {}: driver config must be a mapping",
208            path.display()
209        ));
210    };
211    for key in map.keys() {
212        if key != "profilesDir" {
213            return Err(format!(
214                "{WHO}: {}: unknown config key '{key}'",
215                path.display()
216            ));
217        }
218    }
219    let Some(raw) = map.get("profilesDir") else {
220        return Ok(DriverConfig::default());
221    };
222    let Some(dir) = raw.as_str() else {
223        return Err(format!(
224            "{WHO}: {}: profilesDir must be a string",
225            path.display()
226        ));
227    };
228    // A readable config path always has a parent — "." is the
229    // bare-filename fallback.
230    let base = path.parent().unwrap_or_else(|| Path::new("."));
231    // `~` expands before relativizing — `~/profs` is absolute, a
232    // bare `profs` rebases onto the config file's directory.
233    let expanded = expand_tilde(Path::new(dir), env);
234    let resolved = if expanded.is_absolute() {
235        expanded
236    } else {
237        base.join(expanded)
238    };
239    Ok(DriverConfig {
240        profiles_dir: Some(expand_tilde(&resolved, env)),
241    })
242}

244/// `~`/`~/…` → `$HOME`-joined; anything else unchanged. Only a
245/// leading `~` component expands (same rule scarab's `expand_tilde`
246/// applies for the single-user case).
247fn expand_tilde(path: &Path, env: Env<'_>) -> PathBuf {
248    let mut comps = path.components();
249    if comps.next() != Some(Component::Normal(std::ffi::OsStr::new("~"))) {
250        return path.to_path_buf();
251    }
252    match env("HOME").or_else(|| env("USERPROFILE")) {
253        Some(home) if !home.is_empty() => Path::new(&home).join(comps.as_path()),
254        _ => path.to_path_buf(),
255    }
256}

258/// The `tarsi.*` config file inside `dir`, if any — same preference
259/// order `tarsi-config`'s profile discovery uses (yaml → yml → toml).
260fn config_file_in(dir: &Path) -> Option<PathBuf> {
261    CONFIG_EXTS
262        .iter()
263        .map(|e| dir.join(format!("tarsi.{e}")))
264        .find(|p| p.is_file())
265}

268fn parse_yaml(text: &str) -> Result<Value, String> {

272/// `tamias.yaml|yml` inside `dir` → parsed Value, best-effort.
273fn manifest_in(dir: &Path) -> Option<Value> {
274    for name in ["tamias.yaml", "tamias.yml"] {
275        let p = dir.join(name);
276        if p.is_file() {
277            if let Ok(text) = fs::read_to_string(&p) {
278                if let Ok(v) = parse_yaml(&text) {
279                    return Some(v);
280                }
281            }
282        }
283    }
284    None
285}

287/// Validate `dir` is a well-formed `type: tarsi` profile payload:
288/// a `tarsi.*` file that parses as [`tarsi_core::Config`] through
289/// the real loader (this is what rejects a `[profile]` *table* —
290/// `Config.profile` is the string selector), plus a `tamias.yaml`
291/// whose `type` is `tarsi` when the manifest parses.
292///
293/// Returns `(config_file, manifest)` for the report.
294fn validate_profile(dir: &Path) -> Result<(PathBuf, Option<Value>), String> {
295    let cfg_file = config_file_in(dir).ok_or_else(|| {
296        format!(
297            "{WHO}: {} carries no tarsi.{{yaml,yml,toml}} — not a type:tarsi profile",
298            dir.display()
299        )
300    })?;
301    tarsi_config::load_path(&cfg_file)
302        .map_err(|e| format!("{WHO}: invalid profile config {}: {e}", cfg_file.display()))?;
303    let manifest = manifest_in(dir);
304    if let Some(m) = &manifest {
305        match m.get("type").and_then(Value::as_str) {
306            Some("tarsi") | None => {}
307            Some(other) => {
308                return Err(format!(
309                    "{WHO}: {} declares type '{other}' — this driver serves type: tarsi",
310                    dir.display()
311                ));
312            }
313        }
314    }
315    Ok((cfg_file, manifest))
316}

318/// The dir-profile path for `name` under `root`: `name` →
319/// `<root>/<name>`; `@scope/name` → `<root>/@scope/<name>`.
320/// Rejects anything that isn't a safe relative profile name.
321fn profile_dir_for(root: &Path, name: &str) -> Result<PathBuf, String> {
322    if !safe_profile_name(name) {
323        return Err(format!("{WHO}: unsafe profile name {name:?}"));
324    }
325    let mut out = root.to_path_buf();
326    for seg in name.split('/') {
327        out.push(seg);
328    }
329    Ok(out)
330}

332/// Profile-name safety — mirrors `tarsi-config`'s `safe_name`:
333/// `name` or `@scope/name`, no traversal, no absolute paths, no
334/// NUL. The scope segment must be exactly `@<name>`.
335fn safe_profile_name(name: &str) -> bool {
336    fn seg_ok(s: &str) -> bool {
337        !s.is_empty() && s != "." && s != ".." && !s.contains('\0') && !s.starts_with('@')
338    }
339    if name.is_empty()
340        || name.len() > 128
341        || name.starts_with('/')
342        || name.starts_with('\\')
343        || name.contains('\0')
344    {
345        return false;
346    }
347    let segs: Vec<&str> = name.split('/').collect();
348    match segs.as_slice() {
349        [single] => seg_ok(single),
350        [scope, single] => {
351            scope
352                .strip_prefix('@')
353                .is_some_and(|s| !s.is_empty() && s != "." && s != ".." && !s.contains('\0'))
354                && seg_ok(single)
355        }
356        _ => false,
357    }
358}

360/// The profiles roots `uninstall` scans for managed copies, in
361/// `tarsi-config`'s dir-profile precedence order:
362/// `$TARSI_PROFILES_DIR` then `~/.config/tarsi/profiles`.
363fn profiles_roots(env: Env<'_>) -> Vec<PathBuf> {
364    let mut roots: Vec<PathBuf> = Vec::new();
365    if let Some(d) = env("TARSI_PROFILES_DIR").filter(|s| !s.is_empty()) {
366        roots.push(PathBuf::from(d));
367    }
368    if let Some(home) = env("HOME")
369        .filter(|s| !s.is_empty())
370        .or_else(|| env("USERPROFILE").filter(|s| !s.is_empty()))
371    {
372        roots.push(Path::new(&home).join(".config/tarsi/profiles"));
373    }
374    roots.dedup();
375    roots
376}

378/// Does `entry` name satisfy requested `name`? Mirrors
379/// `tarsi-config`'s `name_matches`: exact, or bare-name match
380/// against a scoped entry (`@scope/name` satisfies `name`, and
381/// `scope/name`/`@scope/name` requests match the scoped entry).
382fn name_matches(req: &str, entry: &str) -> bool {
383    if req == entry {
384        return true;
385    }
386    let bare_req = req.strip_prefix('@').unwrap_or(req);
387    let (esc, epkg) = match entry.strip_prefix('@') {
388        Some(rest) => match rest.split_once('/') {
389            Some((s, p)) => (Some(s), p),
390            None => (None, rest),
391        },
392        None => (None, entry),
393    };
394    match bare_req.split_once('/') {
395        Some((s, p)) => esc == Some(s) && p == epkg,
396        None => epkg == bare_req,
397    }
398}

400/// The bare package name for marker matching — `@scope/name` and
401/// `scope/name` both reduce to `name` (the driver only ever sees
402/// the store record's bare name anyway).
403fn bare_name(name: &str) -> &str {
404    name.rsplit('/').next().unwrap_or(name)
405}

407/// Read the marker file in `dir`, if present and parseable.
408fn read_marker(dir: &Path) -> Option<Value> {

413/// Every marker-bearing profile dir under `root` whose marker name
414/// bare-matches `name` — one level, `@scope/` nesting included
415/// (same walk `tarsi-config`'s `dir_profiles` does).
416fn managed_copies_under(root: &Path, name: &str) -> Vec<PathBuf> {
417    let mut out = Vec::new();
418    let Ok(children) = fs::read_dir(root) else {
419        return out;
420    };
421    for child in children.flatten() {
422        let path = child.path();
423        if !path.is_dir() {
424            continue;
425        }
426        let fname = child.file_name().to_string_lossy().to_string();
427        if fname.starts_with('@') {
428            if let Ok(grand) = fs::read_dir(&path) {
429                for g in grand.flatten() {
430                    let gpath = g.path();
431                    if gpath.is_dir() && marker_matches(&gpath, name) {
432                        out.push(gpath);
433                    }
434                }
435            }
436        } else if marker_matches(&path, name) {
437            out.push(path);
438        }
439    }
440    out
441}

443/// `dir` carries our marker AND the marker's `name` bare-matches
444/// `name`. The marker is the ownership proof — a profile dir
445/// without it is never driver-managed.
446fn marker_matches(dir: &Path, name: &str) -> bool {
447    read_marker(dir)
448        .and_then(|m| m.get("name").and_then(Value::as_str).map(str::to_string))
449        .is_some_and(|n| bare_name(&n) == bare_name(name))
450}

452/// All managed copies of `name` across `roots` (deduped, sorted).
453fn managed_copies(roots: &[PathBuf], name: &str) -> Vec<PathBuf> {
454    let mut out: Vec<PathBuf> = roots
455        .iter()
456        .flat_map(|r| managed_copies_under(r, name))
457        .collect();
458    out.sort();
459    out.dedup();
460    out
461}

463/// Recursively copy `src` → `dst`, skipping the payload's
464/// top-level `drivers/` dir (bundled-driver bits aren't profile
465/// content) and any marker file (a copy of a copy stays clean —
466/// the marker is rewritten by [`sync_profiles_dir`]).
467fn copy_tree(src: &Path, dst: &Path) -> Result<(), String> {
468    copy_inner(src, dst, true)
469}
470
471fn copy_inner(src: &Path, dst: &Path, top: bool) -> Result<(), String> {
472    fs::create_dir_all(dst).map_err(|e| format!("{WHO}: cannot create {}: {e}", dst.display()))?;
473    for entry in
474        fs::read_dir(src).map_err(|e| format!("{WHO}: cannot read {}: {e}", src.display()))?
475    {
476        let entry = entry.map_err(|e| format!("{WHO}: read entry: {e}"))?;
477        let from = entry.path();
478        let name = entry.file_name();
479        let ft = entry
480            .file_type()
481            .map_err(|e| format!("{WHO}: stat {}: {e}", from.display()))?;
482        if top && ft.is_dir() && name == "drivers" {
483            continue;
484        }
485        if name == MARKER {
486            continue;
487        }
488        let to = dst.join(&name);
489        if ft.is_dir() {
490            copy_inner(&from, &to, false)?;
491        } else if ft.is_file() {
492            fs::copy(&from, &to).map_err(|e| {
493                format!(
494                    "{WHO}: cannot copy {} → {}: {e}",
495                    from.display(),
496                    to.display()
497                )
498            })?;
499        }
500        // symlinks/other: skipped — profile content is plain files.
501    }
502    Ok(())
503}

505/// (Re)materialize the `profilesDir` copy of `src` at
506/// `<profilesDir>/<name>/`. An existing dest is replaced only when
507/// it is driver-managed (marker present) or empty of a `tarsi.*` —
508/// a hand-made profile dir is never clobbered.
509fn sync_profiles_dir(
510    src: &Path,
511    profiles_dir: &Path,
512    name: &str,
513    version: Option<&str>,
514) -> Result<PathBuf, String> {
515    let dest = profile_dir_for(profiles_dir, name)?;
516    if dest.exists() {
517        let managed = read_marker(&dest).is_some();
518        let has_profile = config_file_in(&dest).is_some();
519        if !managed && has_profile {
520            return Err(format!(
521                "{WHO}: {} exists and is not driver-managed — refusing to overwrite a hand-made profile",
522                dest.display()
523            ));
524        }
525        fs::remove_dir_all(&dest)
526            .map_err(|e| format!("{WHO}: cannot replace {}: {e}", dest.display()))?;
527    }
528    copy_tree(src, &dest)?;
529    let marker = json!({
530        "driver": "tamias-driver-tarsi",
531        "name": name,
532        "version": version,
533        "source": src,
534    });
535    let text = serde_json::to_string_pretty(&marker)
536        .map_err(|e| format!("{WHO}: cannot serialize marker: {e}"))?;
537    fs::write(dest.join(MARKER), format!("{text}\n"))
538        .map_err(|e| format!("{WHO}: cannot write marker in {}: {e}", dest.display()))?;
539    Ok(dest)
540}

542/// How `name` resolves right now — the `find_dir` pick plus every
543/// visible candidate, classified by source root.
544fn resolve_report(name: &str, env: Env<'_>) -> Value {
545    let resolved = profile::find_dir(name, env);
546    let roots = profiles_roots(env);
547    let source = resolved.as_ref().map(|p| {
548        if roots.iter().any(|r| p.starts_with(r)) {
549            "profiles-dir"
550        } else {
551            "tamias-store"
552        }
553    });
554    let candidates: Vec<Value> = profile::list(env)
555        .iter()
556        .filter(|p| name_matches(name, &p.name))
557        .map(candidate_json)
558        .collect();
559    json!({
560        "name": name,
561        "resolved": resolved,
562        "source": source,
563        "candidates": candidates,
564    })
565}

576/// `install` — validate the payload is a well-formed profile, then
577/// optionally write the `profilesDir` copy. The store commit that
578/// follows this verb is what makes the pack resolvable (and
579/// versioned); see the module docs.
580pub fn install(inv: &Invocation, env: Env<'_>) -> Result<Value, String> {
581    let cfg = load_config(inv.config.as_deref(), env)?;
582    let (cfg_file, manifest) = validate_profile(&inv.dir)?;
583    let version = manifest
584        .as_ref()
585        .and_then(|m| m.get("version"))
586        .and_then(Value::as_str)
587        .map(str::to_string);
588    let mut report = json!({
589        "ok": true,
590        "package": inv.name,
591        "payload": inv.dir,
592        "profile": cfg_file.file_name().map(|n| n.to_string_lossy().to_string()),
593        "manifestVersion": version,
594        "placement": "tamias-store",
595        "note": "the tamias store commit makes this profile resolvable and versioned; no files are copied by default",
596    });
597    if let Some(pd) = &cfg.profiles_dir {
598        let dest = sync_profiles_dir(&inv.dir, pd, &inv.name, version.as_deref())?;
599        report["profilesDir"] = json!(dest);
600        report["placement"] = json!("profiles-dir+store");
601    }
602    Ok(report)
603}
604
605/// `deploy` — refresh operator-facing state for the installed
606/// entry: re-validate it is still a profile payload and re-sync
607/// the `profilesDir` copy when configured, then report how tarsi
608/// resolves the name. Idempotent; never fails on resolution alone.
609pub fn deploy(inv: &Invocation, env: Env<'_>) -> Result<Value, String> {
610    let cfg = load_config(inv.config.as_deref(), env)?;
611    let (cfg_file, _) = validate_profile(&inv.dir)?;
612    let mut report = json!({
613        "ok": true,
614        "package": inv.name,
615        "payload": inv.dir,
616        "profile": cfg_file.file_name().map(|n| n.to_string_lossy().to_string()),
617        "resolution": resolve_report(&inv.name, env),
618    });
619    if let Some(pd) = &cfg.profiles_dir {
620        let version = manifest_in(&inv.dir)
621            .and_then(|m| m.get("version").and_then(Value::as_str).map(str::to_string));
622        let dest = sync_profiles_dir(&inv.dir, pd, &inv.name, version.as_deref())?;
623        report["profilesDir"] = json!(dest);
624    }
625    Ok(report)
626}

628/// `uninstall` — remove driver-managed `profilesDir` copies under
629/// every profiles root (plus the configured one). Only dirs
630/// carrying our marker are touched; the store entry itself is
631/// dropped by tamias after this verb returns.
632pub fn uninstall(inv: &Invocation, env: Env<'_>) -> Result<Value, String> {
633    let cfg = load_config(inv.config.as_deref(), env)?;
634    let mut roots = profiles_roots(env);
635    if let Some(pd) = &cfg.profiles_dir {
636        roots.push(pd.clone());
637    }
638    roots.sort();
639    roots.dedup();
640    let removed = managed_copies(&roots, &inv.name);
641    for dir in &removed {
642        fs::remove_dir_all(dir)
643            .map_err(|e| format!("{WHO}: cannot remove {}: {e}", dir.display()))?;
644    }
645    Ok(json!({
646        "ok": true,
647        "package": inv.name,
648        "removedProfilesDir": removed,
649        "note": "the tamias store entry is dropped by tamias after this verb; unmanaged profile dirs are never touched",
650    }))
651}

698/// Dispatch one tamias verb → its JSON report. `verb` is the
699/// subcommand name; unknown verbs are an error (the clap front in
700/// `main.rs` normally keeps them out).
701pub fn run_verb(verb: &str, inv: &Invocation, env: Env<'_>) -> Result<Value, String> {
702    match verb {
703        "install" => install(inv, env),
704        "deploy" => deploy(inv, env),
705        "uninstall" => uninstall(inv, env),
706        "status" => status(inv, env),
707        "health" => health(inv, env),
708        "logs" => Ok(logs(inv)),
709        "hooks" => Ok(hooks(inv)),
710        other => Err(format!("{WHO}: unknown verb '{other}'")),
711    }
712}

725/// as a JSON value.
726pub fn manifest_value() -> Value {
727    serde_json::to_value(views::manifest()).unwrap_or(Value::Null)

729
730/// Dispatch one parsed pavo wire [`pavo_protocol::wire::Request`] to
731/// its response. `env` feeds the `tarsi-serve` lookup
732/// (`$TARSI_SERVE` override) — injectable so tests never touch the
733/// process env.
734///
735/// Read-only by construction: the only serve ops reachable are
736/// `manifest`/`profiles`/`plan`/`changelog`; `release` is never
737/// dispatched. `render`/`submit`/`jobs`/`notifications` answer
738/// `ok:false` — the manifest claims none of them.
739pub fn dispatch(req: &pavo_protocol::wire::Request, env: Env<'_>) -> pavo_protocol::wire::Response {
740    use pavo_protocol::wire::{QueryResult, Request, Response, ViewSummary};
741    match req {
742        Request::Manifest => Response::ok_typed(&views::manifest()),
743        Request::Views => {
744            let summaries: Vec<ViewSummary> =
745                views::specs().iter().map(ViewSummary::from).collect();
746            Response::ok_typed(&summaries)
747        }
748        Request::View { view, params } => match views::view_result(view, params, env) {
749            Ok(result) => Response::ok_typed(&result),
750            Err(e) => Response::err(e),
751        },
752        Request::Query { query, params } => {
753            if !views::known_query(query) {
754                return Response::err(format!("unknown query: {query}"));
755            }
756            let payload = views::query_data(query, params, env);
757            Response::ok_typed(&QueryResult {
758                query: query.clone(),
759                data: payload,
760            })
761        }
762        Request::Render { kind, .. } => Response::err(format!(
763            "tamias-driver-tarsi claims no renderers (kind '{kind}')"
764        )),
765        Request::Submit { .. } => {
766            Response::err("tamias-driver-tarsi is read-only and implements no forms")
767        }
768        Request::Jobs => Response::err("tamias-driver-tarsi has no job feed"),
769        Request::Notifications => Response::err("tamias-driver-tarsi has no notification feed"),
770    }
771}

773/// The `serve` verb — ndjson JSON-RPC on stdio until EOF or a
774/// broken pipe, via pavo-driver-kit's loop (blank lines skip,
775/// parse failures answer `ok:false`, a dead client exits quietly).
776pub fn serve(env: Env<'_>) -> i32 {
777    let _ = pavo_driver_kit::serve_stdio(|req| dispatch(req, env));
778    0
779}

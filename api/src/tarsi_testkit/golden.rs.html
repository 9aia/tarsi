1//! Golden plan corpus (PL-014 B3) — specs + commit ranges →
2//! expected bump/version/tag/changelog per unit. This is the spec
3//! the legacy TS `tarsi` behavior maps onto: every case is data,
4//! [`GoldenCase::check`] diffs the real [`plan_release`](tarsi_core::plan_release) output
5//! against it and returns human-readable mismatches (empty =
6//! pass).
7//!
8//! Cases:
9//!
10//! - `mixed-single-unit` — one whole-repo unit over
11//!   [`fixtures::SYNTHETIC_COMMITS`]: a breaking `feat!` forces
12//!   major; chore/docs ride along into the changelog.
13//! - `workspace-topo` — two units with a dep edge: path-prefix
14//!   attribution, per-unit bump inference, topo publish order,
15//!   and the unattributed tail (repo-root files).
16//! - `chore-only-none` — docs/chore/test range → `Bump::None`,
17//!   no release, version/tag still computed for the report.
18//! - `real-tarsi-serve-range` — the real `git log` range of
19//!   `crates/tarsi-serve` between scaffold and the D3 driver
20//!   commit (8 commits, newest first as a VCS port returns them).
21//! - `real-myco-range` — the real `git log` range of the myco repo
22//!   between release tags `myco-v0.1.1..myco-v0.1.2` (11 commits,
23//!   whole-repo unit at path `.`).
24//! - `real-tamias-core-range` — the real `git log` range of the
25//!   tamias repo between per-unit tags
26//!   `tamias-core-v0.1.3..tamias-core-v0.1.4` (4 commits, unit path
27//!   `crates/tamias-core` — exercises path-prefix attribution and a
28//!   `release:` noise commit that parses as `Other`).
29//!
30//! Strategy cases (P-030) — `strategy: None` cases are the
31//! `conventional` parity gate; these exercise other strategies:
32//!
33//! - `semantic-release-angular` — the angular commit-analyzer
34//!   preset as data: `revert`→patch joins the bump table,
35//!   BREAKING CHANGES/Features/Bug Fixes/… section headings.
36//! - `custom-deps-strategy` — an inline rule table naming the
37//!   arbitrary raw type `deps` (unreachable via `CommitKind`)
38//!   plus a `fallback` bump for unmatched units.
39//! - `keep-a-changelog` — the KaC builtin: conventional bump
40//!   rules + the six canonical sections; exercises every section
41//!   plus the breaking-routes-to-owning-section veto.

54/// Expected outcome for one unit, in plan (publish) order.
55#[derive(Debug, Clone)]
56pub struct UnitExpect {
57    pub name: String,
58    pub bump: Bump,
59    pub new_version: String,
60    pub tag: String,
61    /// Substrings the rendered changelog must contain.
62    pub changelog_contains: Vec<String>,
63}
64
65/// One golden case: inputs + expectations.
66#[derive(Debug, Clone)]
67pub struct GoldenCase {
68    pub name: &'static str,
69    pub specs: Vec<ReleaseUnitSpec>,
70    /// Newest-first, as the VCS port returns them.
71    pub commits: Vec<Commit>,
72    /// Release strategy (P-030) — `None` = `conventional`, the
73    /// parity baseline every pre-existing case exercises.
74    pub strategy: Option<Strategy>,
75    /// Expected units **in plan order** (topo-sorted).
76    pub expect_units: Vec<UnitExpect>,
77    /// Expected `plan.unattributed` length.
78    pub expect_unattributed: usize,
79}
80
81impl GoldenCase {
82    /// Run the real planner over this case's inputs.
83    ///
84    /// # Errors
85    /// Propagates [`plan_release_with_strategy`] errors (spec/cycle problems).
86    pub fn plan(&self) -> Result<ReleasePlan, tarsi_core::plan::PlanError> {
87        // Fixed date keeps rendered changelogs deterministic.
88        let strategy = self.strategy.clone().unwrap_or_default();
89        plan_release_with_strategy(
90            &self.specs,
91            &self.commits,
92            Some("2026-01-01".to_string()),
93            &strategy,
94        )
95    }
96
97    /// Diff the real plan against expectations — empty `Vec` =
98    /// pass, otherwise one human-readable string per mismatch.
99    #[must_use]
100    pub fn check(&self) -> Vec<String> {
101        let mut bad = Vec::new();
102        let plan = match self.plan() {
103            Ok(p) => p,
104            Err(e) => return vec![format!("plan_release failed: {e}")],
105        };
106        if plan.units.len() != self.expect_units.len() {
107            bad.push(format!(
108                "unit count: got {} ({:?}), want {}",
109                plan.units.len(),
110                plan.units.iter().map(|u| &u.spec.name).collect::<Vec<_>>(),
111                self.expect_units.len()
112            ));
113        }
114        for (i, exp) in self.expect_units.iter().enumerate() {
115            let Some(u) = plan.units.get(i) else {
116                bad.push(format!("missing unit {} at position {i}", exp.name));
117                continue;
118            };
119            if u.spec.name != exp.name {
120                bad.push(format!(
121                    "unit[{i}] name: got {:?}, want {:?}",
122                    u.spec.name, exp.name
123                ));
124                continue;
125            }
126            if u.bump != exp.bump {
127                bad.push(format!(
128                    "{}: bump got {:?}, want {:?}",
129                    exp.name, u.bump, exp.bump
130                ));
131            }
132            if u.new_version.to_string() != exp.new_version {
133                bad.push(format!(
134                    "{}: version got {}, want {}",
135                    exp.name, u.new_version, exp.new_version
136                ));
137            }
138            if u.tag != exp.tag {
139                bad.push(format!(
140                    "{}: tag got {:?}, want {:?}",
141                    exp.name, u.tag, exp.tag
142                ));
143            }
144            let md = u.changelog.render();
145            for needle in &exp.changelog_contains {
146                if !md.contains(needle.as_str()) {
147                    bad.push(format!(
148                        "{}: changelog missing {needle:?}\n---\n{md}\n---",
149                        exp.name
150                    ));
151                }
152            }
153        }
154        if plan.unattributed.len() != self.expect_unattributed {
155            bad.push(format!(
156                "unattributed: got {}, want {}",
157                plan.unattributed.len(),
158                self.expect_unattributed
159            ));
160        }
161        bad
162    }
163}

165fn commit(sha: &str, subject: &str, paths: &[&str]) -> Commit {
166    Commit {
167        sha: sha.into(),
168        subject: subject.into(),
169        body: String::new(),
170        paths: paths.iter().map(|s| (*s).to_string()).collect(),
171    }
172}
173
174fn spec(name: &str, path: &str, version: &str, deps: &[&str]) -> ReleaseUnitSpec {
175    ReleaseUnitSpec {
176        name: name.to_string(),
177        path: path.to_string(),
178        current_version: Version::parse(version).expect("golden spec version"),
179        tag_format: tarsi_core::unit::DEFAULT_TAG_FORMAT.to_string(),
180        bump: None,
181        publish: vec![PublishTarget::CratesIo, PublishTarget::GithubRelease],
182        depends_on: deps.iter().map(|s| (*s).to_string()).collect(),
183    }
184}
185
186fn unit(
187    name: &str,
188    bump: Bump,
189    new_version: &str,
190    tag: &str,
191    changelog_contains: &[&str],
192) -> UnitExpect {
193    UnitExpect {
194        name: name.to_string(),
195        bump,
196        new_version: new_version.to_string(),
197        tag: tag.to_string(),
198        changelog_contains: changelog_contains
199            .iter()
200            .map(|s| (*s).to_string())
201            .collect(),
202    }
203}

1//! `[release.sync_refs]` — pack-ref/doc version sync (T-039).
2//!
3//! Repos used to do this in `scripts/release.sh` *before* calling
4//! `tarsi-serve release`: learn the planned version, `sed`
5//! `name@old` → `name@new` in `libraries/*/tamias.yaml` +
6//! README/SKILL install examples, commit, *then* release — so the
7//! tagged tree carried consistent refs. This stage folds that into
8//! the pipeline: it runs inside `bump`/`release` after manifests
9//! are bumped and before the release commit, so synced files ride
10//! the same commit the tag points at.
11//!
12//! Safety model (mirrors the shims):
13//!
14//! - **Opt-in** — no `[release.sync_refs]` rules, no-op.
15//! - **Single-version only** — sync needs one unambiguous
16//!   `old → new` pair. A plan whose releasing units disagree on
17//!   `new_version` (or `current_version`) is skipped with a
18//!   reason, exactly like the shims' `len(vs)==1` python guard.
19//! - **Idempotent** — rules match the *old* version; once refs
20//!   say `new`, nothing matches and the stage reports zero
21//!   changes. Re-running `bump`/`release` after a successful sync
22//!   is a no-op for this stage.
23//! - **Dry-run safe** — `dry_run: true` computes the same report
24//!   (files + replacement counts) without writing.
25
26use std::collections::BTreeMap;
27use std::path::{Path, PathBuf};
28
29use regex::Regex;
30use serde_json::{json, Value};
31use tarsi_core::config::{SyncRefRule, SyncRefsConfig};

33/// What the stage did (or would do). Serialized into the `bump`
34/// result's `sync_refs` key and the `release` result's.
35#[derive(Debug, Clone)]
36pub struct SyncOutcome {
37    /// The report value embedded in op results.
38    pub report: Value,
39    /// Repo-root-relative paths whose content changed (or would
40    /// change under dry-run) — the caller stages these into the
41    /// release commit.
42    pub changed_files: Vec<String>,
43}
44
45/// Resolve the single `old → new` version pair the rules rewrite
46/// between. Releasing units are plan-unit JSON carrying
47/// `spec.current_version` and `new_version`. `Err(reason)` reports
48/// a skip condition, not a failure — the release proceeds without
49/// syncing, like the shims.
50fn version_pair(releasing: &[Value]) -> Result<(String, String), String> {
51    let mut olds = std::collections::BTreeSet::new();
52    let mut news = std::collections::BTreeSet::new();
53    for u in releasing {
54        olds.insert(
55            u["spec"]["current_version"]
56                .as_str()
57                .unwrap_or_default()
58                .to_string(),
59        );
60        news.insert(u["new_version"].as_str().unwrap_or_default().to_string());
61    }
62    if olds.len() != 1 || news.len() != 1 {
63        return Err(format!(
64            "mixed-version plan ({} old × {} new across {} releasing units)",
65            olds.len(),
66            news.len(),
67            releasing.len()
68        ));
69    }
70    let old = olds.into_iter().next().unwrap_or_default();
71    let new = news.into_iter().next().unwrap_or_default();
72    if old == new {
73        return Err("no version change".to_string());
74    }
75    Ok((old, new))
76}

78/// Compile one rule's `find` template: `{old}`/`{new}` expand to
79/// the regex-escaped versions so a literal `1.2.3` can't match
80/// `1.2.30`. `replace` stays a literal template expanded later.
81fn compile_find(rule: &SyncRefRule, idx: usize, old: &str, new: &str) -> Result<Regex, String> {
82    let pat = rule
83        .find
84        .replace("{old}", &regex::escape(old))
85        .replace("{new}", &regex::escape(new));
86    Regex::new(&pat).map_err(|e| format!("sync_refs rule {idx}: bad find regex {pat:?}: {e}"))
87}
88
89/// The literal replacement for one rule — `{old}`/`{new}` expand
90/// unescaped (replacement text is literal, not regex).
91fn expand_replace(rule: &SyncRefRule, old: &str, new: &str) -> String {
92    rule.replace.replace("{old}", old).replace("{new}", new)
93}
94
95/// Expand a rule's `files` globs to concrete files under `dir`,
96/// sorted + deduped. Non-magic patterns that don't exist and
97/// magic patterns matching nothing both yield nothing — missing
98/// files are not an error (a repo may carry rules for files it
99/// doesn't have).
100fn expand_files(dir: &Path, files: &[String]) -> Result<Vec<PathBuf>, String> {
101    let mut out = Vec::new();
102    for pat in files {
103        let full = dir.join(pat);
104        let full_str = full.to_string_lossy().to_string();
105        match glob::glob(&full_str) {
106            Ok(paths) => {
107                for p in paths.flatten() {
108                    if p.is_file() {
109                        out.push(p);
110                    }
111                }
112            }
113            Err(e) => return Err(format!("sync_refs: bad file glob {pat:?}: {e}")),
114        }
115    }
116    out.sort();
117    out.dedup();
118    Ok(out)
119}

121/// Repo-relative `/`-separated path for staging/reporting.
122fn rel_path(dir: &Path, p: &Path) -> String {
123    p.strip_prefix(dir)
124        .unwrap_or(p)
125        .to_string_lossy()
126        .replace('\\', "/")
127}

129/// Run the sync stage. `releasing` is the plan-unit JSON array the
130/// ops already carry; `dry_run` computes the report without
131/// writing. Never fails on *skip* conditions (mixed versions, no
132/// rules) — only on I/O or bad config, which the caller turns into
133/// a step error.
134pub fn run(
135    dir: &Path,
136    cfg: &SyncRefsConfig,
137    releasing: &[Value],
138    dry_run: bool,
139) -> Result<SyncOutcome, String> {
140    if cfg.rules.is_empty() {
141        return Ok(SyncOutcome {
142            report: json!({ "status": "disabled" }),
143            changed_files: Vec::new(),
144        });
145    }
146    let (old, new) = match version_pair(releasing) {
147        Ok(pair) => pair,
148        Err(reason) => {
149            return Ok(SyncOutcome {
150                report: json!({
151                    "status": "skipped",
152                    "reason": reason,
153                    "old": Value::Null,
154                    "new": Value::Null,
155                    "changes": [],
156                }),
157                changed_files: Vec::new(),
158            });
159        }
160    };
161
162    // Compile all rules up front — a bad regex fails before any
163    // file is touched.
164    let compiled: Vec<(usize, Regex, String)> = cfg
165        .rules
166        .iter()
167        .enumerate()
168        .map(|(i, r)| {
169            compile_find(r, i, &old, &new).map(|re| (i, re, expand_replace(r, &old, &new)))
170        })
171        .collect::<Result<_, _>>()?;
172
173    // Apply rules in order over an in-memory overlay so later
174    // rules see earlier rules' rewrites (and dry-run reports what
175    // a real run would do). Writes happen once, at the end.
176    let mut contents: BTreeMap<PathBuf, String> = BTreeMap::new();
177    let mut changes: Vec<Value> = Vec::new();
178    let mut changed_files: Vec<String> = Vec::new();
179    for (rule_idx, re, repl) in &compiled {
180        let rule = &cfg.rules[*rule_idx];
181        for path in expand_files(dir, &rule.files)? {
182            let text = match contents.get(&path) {
183                Some(t) => t.clone(),
184                None => std::fs::read_to_string(&path)
185                    .map_err(|e| format!("sync_refs: read {}: {e}", path.display()))?,
186            };
187            let n = re.find_iter(&text).count();
188            if n == 0 {
189                continue;
190            }
191            let new_text = re
192                .replace_all(&text, |_: &regex::Captures| repl.as_str())
193                .into_owned();
194            if new_text == text {
195                continue;
196            }
197            changes.push(json!({
198                "rule": rule_idx,
199                "file": rel_path(dir, &path),
200                "replacements": n,
201            }));
202            let rel = rel_path(dir, &path);
203            if !changed_files.contains(&rel) {
204                changed_files.push(rel);
205            }
206            contents.insert(path, new_text);
207        }
208    }
209
210    if !dry_run {
211        for (path, text) in &contents {
212            std::fs::write(path, text)
213                .map_err(|e| format!("sync_refs: write {}: {e}", path.display()))?;
214        }
215    }
216
217    Ok(SyncOutcome {
218        report: json!({
219            "status": if dry_run { "dry-run" } else { "applied" },
220            "reason": Value::Null,
221            "old": old,
222            "new": new,
223            "changes": changes,
224        }),
225        changed_files,
226    })
227}

234    fn unit_json(current: &str, new: &str) -> Value {

241    fn rule(find: &str, replace: &str, files: &[&str]) -> SyncRefRule {
242        SyncRefRule {
243            files: files.iter().map(|s| (*s).to_string()).collect(),
244            find: find.to_string(),
245            replace: replace.to_string(),
246        }
247    }

249    #[test]
250    fn pair_requires_single_old_and_new() {
251        assert_eq!(
252            version_pair(&[unit_json("0.1.0", "0.2.0")]).unwrap(),
253            ("0.1.0".to_string(), "0.2.0".to_string())
254        );
255        assert!(version_pair(&[unit_json("0.1.0", "0.2.0"), unit_json("0.1.0", "0.1.1")]).is_err());
256        assert!(version_pair(&[unit_json("0.1.0", "0.1.0")]).is_err());
257        assert!(version_pair(&[]).is_err());

279    #[test]
280    fn rewrites_refs_and_is_idempotent() {
281        let d = tempfile::tempdir().unwrap();
282        std::fs::write(
283            d.path().join("README.md"),
284            "install: tamias install app@0.1.0\napp@0.1.0 again\n",
285        )
286        .unwrap();
287        let cfg = SyncRefsConfig {
288            rules: vec![rule("app@{old}", "app@{new}", &["README.md"])],
289        };
290        let releasing = vec![unit_json("0.1.0", "0.2.0")];
291        let out = run(d.path(), &cfg, &releasing, false).unwrap();
292        assert_eq!(out.report["status"], "applied");
293        assert_eq!(out.changed_files, vec!["README.md".to_string()]);
294        assert_eq!(out.report["changes"][0]["replacements"], 2);
295        let text = std::fs::read_to_string(d.path().join("README.md")).unwrap();
296        assert_eq!(text, "install: tamias install app@0.2.0\napp@0.2.0 again\n");
297        // Second run: nothing left matching `old` → zero changes.
298        let out2 = run(d.path(), &cfg, &releasing, false).unwrap();
299        assert!(out2.changed_files.is_empty());
300        assert_eq!(out2.report["changes"].as_array().unwrap().len(), 0);
301    }

303    #[test]
304    fn version_is_regex_escaped_but_not_bounded() {
305        // `{old}` expands regex-escaped (the `.`s are literal), but
306        // — like the shims' `sed s|name@old|…|g` — the pattern is
307        // not auto-bounded: `app@1.2.3` still matches the prefix of
308        // `app@1.2.30`. Refs that need a boundary carry one in
309        // `find` (e.g. `app@{old}\b` — `regex` has no look-around;
310        // `\b` suffices since versions are word chars).
311        let d = tempfile::tempdir().unwrap();
312        std::fs::write(d.path().join("r.md"), "app@1.2.30 app@1.2.3\n").unwrap();
313        let cfg = SyncRefsConfig {
314            rules: vec![rule("app@{old}\\b", "app@{new}", &["r.md"])],
315        };
316        let releasing = vec![unit_json("1.2.3", "1.2.4")];
317        run(d.path(), &cfg, &releasing, false).unwrap();
318        assert_eq!(
319            std::fs::read_to_string(d.path().join("r.md")).unwrap(),
320            "app@1.2.30 app@1.2.4\n"
321        );
322    }

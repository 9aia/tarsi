1//! tarsi-manifest — the filesystem edge of unit resolution.
2//! Reads `Cargo.toml` (workspace members, package versions,
3//! path dependencies) and produces [`ReleaseUnitSpec`]s for
4//! [`tarsi_core::plan_release`]. Also owns manifest version
5//! writes (`cargo set-version` equivalent) via `toml_edit` so
6//! formatting/comments survive.
7//!
8//! Pure parsing lives here; this is the only tarsi crate allowed
9//! to touch `Cargo.toml` files.
10
11use std::path::{Path, PathBuf};
12
13use semver::Version;
14use tarsi_core::{PublishTarget, ReleaseUnitSpec};

16#[derive(Debug, thiserror::Error)]
17pub enum ManifestError {
18    #[error("{0}: {1}")]
19    Io(PathBuf, std::io::Error),
20    #[error("{0}: toml: {1}")]
21    Toml(PathBuf, String),
22    #[error("{0}: no [package] name/version")]
23    NoPackage(PathBuf),
24    #[error("workspace member glob {0:?} matched nothing")]
25    EmptyGlob(String),
26}
27
28fn read_toml(path: &Path) -> Result<toml::Table, ManifestError> {
29    let text =
30        std::fs::read_to_string(path).map_err(|e| ManifestError::Io(path.to_path_buf(), e))?;
31    toml::from_str(&text).map_err(|e| ManifestError::Toml(path.to_path_buf(), e.to_string()))
32}
33
34fn err_io(p: &Path, e: std::io::Error) -> ManifestError {
35    ManifestError::Io(p.to_path_buf(), e)
36}

38/// Workspace members declared in `<root>/Cargo.toml`
39/// (`workspace.members`, default `["."]`… actually cargo requires
40/// members; empty → root package only).
41fn member_globs(root: &Path) -> Result<Vec<String>, ManifestError> {
42    let manifest = root.join("Cargo.toml");
43    let table = read_toml(&manifest)?;
44    let Some(workspace) = table.get("workspace").and_then(|w| w.as_table()) else {
45        // Single-package repo — the root IS the unit.
46        return Ok(vec![".".to_string()]);
47    };
48    let Some(members) = workspace.get("members").and_then(|m| m.as_array()) else {
49        return Ok(vec![".".to_string()]);
50    };
51    Ok(members
52        .iter()
53        .filter_map(|m| m.as_str().map(str::to_string))
54        .collect())
55}

57/// Expand member globs (`crates/*`, `crates/foo`) to concrete dirs.
58fn expand_members(root: &Path, globs: &[String]) -> Result<Vec<PathBuf>, ManifestError> {
59    let mut dirs = Vec::new();
60    for g in globs {
61        if g.contains('*') {
62            let (base, _) = g.split_once('*').unwrap_or((g, ""));
63            let base_path = root.join(base.trim_end_matches('/'));
64            let rd = std::fs::read_dir(&base_path).map_err(|e| err_io(&base_path, e))?;
65            let mut any = false;
66            for entry in rd.flatten() {
67                let p = entry.path();
68                if p.is_dir() && p.join("Cargo.toml").exists() {
69                    any = true;
70                    dirs.push(p);
71                }
72            }
73            if !any {
74                return Err(ManifestError::EmptyGlob(g.clone()));
75            }
76        } else {
77            let p = root.join(g);
78            if p.join("Cargo.toml").exists() {
79                dirs.push(p);
80            }
81        }
82    }
83    dirs.sort();
84    dirs.dedup();
85    Ok(dirs)
86}

88struct MemberInfo {
89    name: String,
90    version: Version,
91    /// Version resolves from `[workspace.package].version`
92    /// (`version.workspace = true`, or the inherited fallback) —
93    /// such units share one version file and must release in
94    /// lockstep (I-002). Literal per-package versions are
95    /// independent.
96    version_shared: bool,
97    publishable: bool,
98    /// Names of sibling members this one path-depends on.
99    path_deps: Vec<String>,
100}
101
102fn read_member(dir: &Path, ws_version: Option<&Version>) -> Result<MemberInfo, ManifestError> {
103    let manifest = dir.join("Cargo.toml");
104    let table = read_toml(&manifest)?;
105    let package = table
106        .get("package")
107        .and_then(|p| p.as_table())
108        .ok_or_else(|| ManifestError::NoPackage(manifest.clone()))?;
109    let name = package
110        .get("name")
111        .and_then(|n| n.as_str())
112        .ok_or_else(|| ManifestError::NoPackage(manifest.clone()))?
113        .to_string();
114    // `version.workspace = true` inherits workspace.package.version.
115    // Anything that isn't a literal string (the inherit table, or
116    // a missing key falling back to the workspace version) shares
117    // the workspace version file → `version_shared`.
118    let version_item = package.get("version");
119    let version_shared = !version_item.is_some_and(|v| v.is_str());
120    let version = match version_item {
121        Some(v) if v.is_str() => Version::parse(v.as_str().unwrap_or("0.0.0"))
122            .map_err(|e| ManifestError::Toml(manifest.clone(), e.to_string()))?,
123        Some(v) if v.is_table() => ws_version
124            .cloned()
125            .ok_or_else(|| ManifestError::NoPackage(manifest.clone()))?,
126        _ => ws_version
127            .cloned()
128            .ok_or_else(|| ManifestError::NoPackage(manifest.clone()))?,
129    };
130    let publishable = package
131        .get("publish")
132        .and_then(|p| p.as_bool())
133        .unwrap_or(true);
134    // Path deps → sibling unit names. Deps can rename the package
135    // (`foo = { package = "real-name", path = "../x" }`).
136    let mut path_deps = Vec::new();
137    for section in ["dependencies", "build-dependencies"] {
138        if let Some(deps) = table.get(section).and_then(|d| d.as_table()) {
139            for (dep_name, spec) in deps {
140                // Only table specs can carry `path = "…"`.
141                let Some(t) = spec.as_table() else { continue };
142                if t.get("path").is_none() {
143                    continue;
144                }
145                let real = t
146                    .get("package")
147                    .and_then(|p| p.as_str())
148                    .unwrap_or(dep_name);
149                path_deps.push(real.to_string());
150            }
151        }
152    }
153    // Workspace-inherited deps (`workspace = true`) — resolve names
154    // from [workspace.dependencies] path entries.
155    if let Some(ws_deps) = table
156        .get("workspace")
157        .and_then(|w| w.as_table())
158        .and_then(|w| w.get("dependencies"))
159        .and_then(|d| d.as_table())
160    {
161        for section in ["dependencies", "build-dependencies"] {
162            if let Some(deps) = table.get(section).and_then(|d| d.as_table()) {
163                for (dep_name, spec) in deps {
164                    if spec
165                        .as_table()
166                        .is_some_and(|t| t.get("workspace") == Some(&toml::Value::Boolean(true)))
167                    {
168                        if let Some(real) = ws_deps
169                            .get(dep_name)
170                            .and_then(|d| d.as_table())
171                            .filter(|t| t.get("path").is_some())
172                            .and_then(|t| t.get("package"))
173                            .and_then(|p| p.as_str())
174                            .or(Some(dep_name.as_str()))
175                        {
176                            path_deps.push(real.to_string());
177                        }
178                    }
179                }
180            }
181        }
182    }
183    path_deps.sort();
184    path_deps.dedup();
185    Ok(MemberInfo {
186        name,
187        version,
188        version_shared,
189        publishable,
190        path_deps,
191    })
192}

194/// Scan `<root>` (a cargo workspace or single package) into
195/// [`ReleaseUnitSpec`]s — versions, paths, dep topology, and
196/// publish targets (`publish = false` manifests → no targets).
197///
198/// `tag_format`/`bump`/`publish` come from config overrides
199/// applied by the caller ([`apply_config`]).
200pub fn cargo_workspace_units(root: &Path) -> Result<Vec<ReleaseUnitSpec>, ManifestError> {
201    let root = root.canonicalize().map_err(|e| err_io(root, e))?;
202    let globs = member_globs(&root)?;
203    let dirs = expand_members(&root, &globs)?;
204    // workspace.package.version for `version.workspace = true`.
205    let ws_version = read_toml(&root.join("Cargo.toml")).ok().and_then(|t| {
206        t.get("workspace")
207            .and_then(|w| w.as_table())
208            .and_then(|w| w.get("package"))
209            .and_then(|p| p.as_table())
210            .and_then(|p| p.get("version"))
211            .and_then(|v| v.as_str())
212            .and_then(|s| Version::parse(s).ok())
213    });
214    let mut members = Vec::new();
215    for dir in &dirs {
216        members.push((dir.clone(), read_member(dir, ws_version.as_ref())?));
217    }
218    // Map package name → spec; path deps become depends_on filtered
219    // to known members.
220    let names: std::collections::BTreeSet<String> =
221        members.iter().map(|(_, m)| m.name.clone()).collect();
222    let mut specs = Vec::with_capacity(members.len());
223    for (dir, m) in members {
224        let rel = dir
225            .strip_prefix(&root)
226            .unwrap_or(&dir)
227            .to_string_lossy()
228            .replace('\\', "/");
229        let path = if rel.is_empty() { ".".to_string() } else { rel };
230        specs.push(ReleaseUnitSpec {
231            name: m.name.clone(),
232            path,
233            current_version: m.version,
234            tag_format: "{unit}-v{version}".to_string(),
235            bump: None,
236            publish: if m.publishable {
237                vec![PublishTarget::CratesIo, PublishTarget::GithubRelease]
238            } else {
239                Vec::new()
240            },
241            depends_on: m
242                .path_deps
243                .into_iter()
244                .filter(|d| names.contains(d))
245                .collect(),
246            version_shared: m.version_shared,
247        });
248    }
249    if specs.is_empty() {
250        return Err(ManifestError::EmptyGlob("*".to_string()));
251    }
252    Ok(specs)
253}

261/// Write `version` into `<manifest>/Cargo.toml` `[package]`,
262/// preserving the rest of the file (toml_edit). Returns true when
263/// the file changed.
264pub fn write_version(manifest: &Path, version: &Version) -> Result<bool, ManifestError> {
265    let path = manifest.join("Cargo.toml");
266    let text = std::fs::read_to_string(&path).map_err(|e| err_io(&path, e))?;
267    let mut doc = text
268        .parse::<toml_edit::DocumentMut>()
269        .map_err(|e| ManifestError::Toml(path.clone(), e.to_string()))?;
270    let pkg = doc
271        .get_mut("package")
272        .and_then(|p| p.as_table_mut())
273        .ok_or_else(|| ManifestError::NoPackage(path.clone()))?;
274    let item = pkg
275        .get_mut("version")
276        .ok_or_else(|| ManifestError::NoPackage(path.clone()))?;
277    // `version.workspace = true` can't be written per-package —
278    // caller must handle workspace-versioned packages separately.
279    if item.is_str() {
280        let new = version.to_string();
281        if item.as_str() == Some(new.as_str()) {
282            return Ok(false);
283        }
284        *item = toml_edit::value(new);
285        std::fs::write(&path, doc.to_string()).map_err(|e| err_io(&path, e))?;
286        return Ok(true);
287    }
288    Err(ManifestError::Toml(
289        path,
290        "version is workspace-inherited; bump workspace.package.version instead".to_string(),
291    ))
292}

294/// Write `version` into `[workspace.package]` when members inherit
295/// it — used when every releasing unit shares one version.
296pub fn write_workspace_version(root: &Path, version: &Version) -> Result<bool, ManifestError> {
297    let path = root.join("Cargo.toml");
298    let text = std::fs::read_to_string(&path).map_err(|e| err_io(&path, e))?;
299    let mut doc = text
300        .parse::<toml_edit::DocumentMut>()
301        .map_err(|e| ManifestError::Toml(path.clone(), e.to_string()))?;
302    let Some(pkg) = doc
303        .get_mut("workspace")
304        .and_then(|w| w.as_table_mut())
305        .and_then(|w| w.get_mut("package"))
306        .and_then(|p| p.as_table_mut())
307    else {
308        return Ok(false);
309    };
310    let Some(item) = pkg.get_mut("version") else {
311        return Ok(false);
312    };
313    if !item.is_str() {
314        return Ok(false);
315    }
316    let new = version.to_string();
317    if item.as_str() == Some(new.as_str()) {
318        return Ok(false);
319    }
320    *item = toml_edit::value(new);
321    std::fs::write(&path, doc.to_string()).map_err(|e| err_io(&path, e))?;
322    Ok(true)
323}

325/// Per-unit version write that handles `version.workspace = true`:
326/// if the member inherits, bump the workspace version instead
327/// (only correct when all releasing units share it — callers check).
328pub fn write_unit_version(
329    root: &Path,
330    unit_path: &str,
331    version: &Version,
332) -> Result<bool, ManifestError> {
333    let manifest = root.join(unit_path);
334    let path = manifest.join("Cargo.toml");
335    let text = std::fs::read_to_string(&path).map_err(|e| err_io(&path, e))?;
336    let doc: toml::Table =
337        toml::from_str(&text).map_err(|e| ManifestError::Toml(path.clone(), e.to_string()))?;
338    let inherits = doc
339        .get("package")
340        .and_then(|p| p.as_table())
341        .and_then(|p| p.get("version"))
342        .and_then(|v| v.as_table())
343        .is_some_and(|t| t.get("workspace") == Some(&toml::Value::Boolean(true)));
344    if inherits {
345        return write_workspace_version(root, version);
346    }
347    write_version(&manifest, version)
348}

350/// Navigate `doc` along `path` of table keys to a [`TableLike`].
351fn table_at<'a>(
352    doc: &'a mut toml_edit::DocumentMut,
353    path: &[&str],
354) -> Option<&'a mut dyn toml_edit::TableLike> {
355    let mut item = doc.as_item_mut();
356    for key in path {
357        item = item.as_table_like_mut()?.get_mut(key)?;
358    }
359    item.as_table_like_mut()
360}

362/// The version a dep's manifest currently declares — literal
363/// `[package].version`, or `[workspace.package].version` when the
364/// package inherits (`version.workspace = true`).
365fn dep_manifest_version(manifest: &Path, ws_version: Option<&str>) -> Option<String> {
366    let text = std::fs::read_to_string(manifest).ok()?;
367    let doc: toml::Table = toml::from_str(&text).ok()?;
368    let v = doc.get("package")?.get("version")?;
369    if let Some(s) = v.as_str() {
370        return Some(s.to_string());
371    }
372    if v.as_table()
373        .is_some_and(|t| t.get("workspace") == Some(&toml::Value::Boolean(true)))
374    {
375        return ws_version.map(str::to_string);
376    }
377    None
378}

380/// Rewrite `version` reqs on path deps so each names the target
381/// package's *current* manifest version. `cargo publish` ships path
382/// deps with their `version` req — after a bump the req must match
383/// the new version (`^0.1.5` rejects `0.2.0`, and cargo's own
384/// resolver fails the same way on `cargo check`). Run after
385/// [`write_unit_version`] writes land. Covers
386/// `[workspace.dependencies]` in the root manifest plus
387/// `dependencies`/`build-dependencies`/`dev-dependencies` in every
388/// workspace member; only entries carrying both `path` and a string
389/// `version` are touched. Returns true when any file changed.
390pub fn sync_path_dep_reqs(root: &Path) -> Result<bool, ManifestError> {
391    let root = root.canonicalize().map_err(|e| err_io(root, e))?;
392    let ws_version = read_toml(&root.join("Cargo.toml")).ok().and_then(|t| {
393        t.get("workspace")?
394            .get("package")?
395            .get("version")?
396            .as_str()
397            .map(str::to_string)
398    });
399    let mut manifests = vec![(root.join("Cargo.toml"), true)];
400    let globs = member_globs(&root)?;
401    for dir in expand_members(&root, &globs)? {
402        manifests.push((dir.join("Cargo.toml"), false));
403    }
404    let mut changed = false;
405    for (manifest, is_root) in manifests {
406        if !manifest.exists() {
407            continue;
408        }
409        let Some(base) = manifest.parent() else {
410            continue;
411        };
412        let sections: &[&[&str]] = if is_root {
413            &[&["workspace", "dependencies"]]
414        } else {
415            &[
416                &["dependencies"],
417                &["build-dependencies"],
418                &["dev-dependencies"],
419            ]
420        };
421        let text = std::fs::read_to_string(&manifest).map_err(|e| err_io(&manifest, e))?;
422        let mut doc = text
423            .parse::<toml_edit::DocumentMut>()
424            .map_err(|e| ManifestError::Toml(manifest.clone(), e.to_string()))?;
425        let mut file_changed = false;
426        for section in sections {
427            let Some(table) = table_at(&mut doc, section) else {
428                continue;
429            };
430            let entries: Vec<(String, String, String, String)> = table
431                .iter()
432                .filter_map(|(key, item)| {
433                    let t = item.as_table_like()?;
434                    let path = t.get("path")?.as_str()?.to_string();
435                    let req = t.get("version")?.as_str()?.to_string();
436                    let name = t
437                        .get("package")
438                        .and_then(|p| p.as_str())
439                        .unwrap_or(key)
440                        .to_string();
441                    Some((key.to_string(), path, name, req))
442                })
443                .collect();
444            for (key, dep_path, _name, req) in entries {
445                let dep_manifest = base.join(&dep_path).join("Cargo.toml");
446                let Some(actual) = dep_manifest_version(&dep_manifest, ws_version.as_deref())
447                else {
448                    continue;
449                };
450                if actual == req {
451                    continue;
452                }
453                let Some(spec) = table
454                    .get_mut(&key)
455                    .and_then(|i| i.as_table_like_mut())
456                    .and_then(|t| t.get_mut("version"))
457                else {
458                    continue;
459                };
460                *spec = toml_edit::value(actual);
461                file_changed = true;
462            }
463        }
464        if file_changed {
465            std::fs::write(&manifest, doc.to_string()).map_err(|e| err_io(&manifest, e))?;
466            changed = true;
467        }
468    }
469    Ok(changed)
470}

599                    "[package]\nname = \"core\"\nversion.workspace = true\n"
600                },
601            )
602            .unwrap();
603        }
604        // Bump the workspace version, then sync — the req must follow.
605        write_unit_version(root, "crates/cli", &Version::parse("2.0.0").unwrap()).unwrap();
606        assert!(sync_path_dep_reqs(root).unwrap());
607        let text = std::fs::read_to_string(root.join("Cargo.toml")).unwrap();
608        assert!(
609            text.contains("core = { path = \"crates/core\", version = \"2.0.0\" }"),
610            "{text}"
611        );
612        // Idempotent — second run changes nothing.
613        assert!(!sync_path_dep_reqs(root).unwrap());
614    }

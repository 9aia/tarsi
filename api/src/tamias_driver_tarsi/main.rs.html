1//! `tamias-driver-tarsi` — dual-surface driver: the tamias content
2//! driver for `type: tarsi` release-profile packs (PL-014 D3) AND
3//! the pavo read-only view driver over tarsi release state. One
4//! binary, two consumers, disjoint argv — see `tarsi_driver`'s
5//! module docs for both contracts.
6//!
7//! pavo surface (verbs take no `--dir`/`--name`):
8//!
9//! - `manifest` — print the `pavo_protocol::DriverManifest` JSON
10//!   (discovery probe; never touches tarsi state).
11//! - `capabilities` — merged JSON advertising BOTH surfaces: pavo's
12//!   `{"capabilities": ["pavo",…], "verbs": […"serve"…]}` shape plus
13//!   tamias's `"types": ["tarsi"]` and the content verbs in
14//!   `verbs` — one payload satisfies both discoverers.
15//! - `serve` — ndjson JSON-RPC on stdio (`manifest`/`views`/`view`/
16//!   `query`); queries ride `tarsi-serve` one-shot ops (plan
17//!   preview, changelog preview, profile list). `$TARSI_SERVE`
18//!   overrides the `tarsi-serve` binary (default: PATH).
19//!
20//! ```text
21//! tamias-driver-tarsi manifest|capabilities|serve
22//! tamias-driver-tarsi <verb> --dir <d> --name <n> [--config <f>]
23//! ```
24//!
25//! `verb ∈ {config-schema, install, deploy, uninstall, status,
26//! health, logs, hooks}`. tamias dispatches the lifecycle verbs
27//! with `--dir`/`--name` always present; the probe verbs take no
28//! flags. Every verb answers a JSON report on stdout; failures go
29//! to stderr with exit 1.
30

33use std::path::PathBuf;
34use std::process::ExitCode;

36use tarsi_driver as drv;

38const USAGE: &str = "usage: tamias-driver-tarsi manifest|capabilities|serve\n       tamias-driver-tarsi <verb> --dir <d> --name <n> [--config <f>]\n       verbs: capabilities config-schema install deploy uninstall status health logs hooks";

40/// The flags tamias appends to every dispatched verb.
41#[derive(Debug, Default)]
42struct Args {
43    dir: Option<PathBuf>,
44    name: Option<String>,
45    config: Option<PathBuf>,
46}
47
48/// Parse the trailing argv after the verb: `--dir`/`--name`/
49/// `--config` in `--flag v` or `--flag=v` form. Unknown flags and
50/// stray positionals are errors — the wire is exact.
51fn parse_args(args: &[String]) -> Result<Args, String> {
52    let mut out = Args::default();
53    let mut i = 0;
54    while i < args.len() {
55        let a = args[i].as_str();
56        let (flag, inline) = match a.split_once('=') {
57            Some((f, v)) if f.starts_with("--") => (f, Some(v.to_string())),
58            _ => (a, None),
59        };
60        let take = |i: &mut usize| -> Result<String, String> {
61            if let Some(v) = &inline {
62                return Ok(v.clone());
63            }
64            *i += 1;
65            args.get(*i)
66                .cloned()
67                .ok_or_else(|| format!("tamias-driver-tarsi: {flag} needs a value"))
68        };
69        match flag {
70            "--dir" => out.dir = Some(PathBuf::from(take(&mut i)?)),
71            "--name" => out.name = Some(take(&mut i)?),
72            "--config" => out.config = Some(PathBuf::from(take(&mut i)?)),
73            "--help" | "-h" => return Err("help".to_string()),
74            other => return Err(format!("tamias-driver-tarsi: unknown flag '{other}'")),
75        }
76        i += 1;
77    }
78    Ok(out)
79}

81fn env() -> impl Fn(&str) -> Option<String> {
82    |k| std::env::var(k).ok()
83}
84
85fn main() -> ExitCode {
86    let args: Vec<String> = std::env::args().skip(1).collect();
87    let Some(verb) = args.first().map(String::as_str) else {
88        eprintln!("{USAGE}");
89        return ExitCode::from(2);
90    };
91    match verb {
92        // pavo probes — bare argv, no --dir/--name. `capabilities`
93        // answers the merged pavo+tamias payload.
94        "manifest" => {
95            drv::emit(&drv::manifest_value());
96            ExitCode::SUCCESS
97        }
98        "capabilities" => {
99            drv::emit(&drv::capabilities_value());
100            ExitCode::SUCCESS
101        }
102        "serve" => ExitCode::from(drv::serve(&env()) as u8),
103        "config-schema" => {
104            drv::emit(&drv::config_schema());
105            ExitCode::SUCCESS
106        }
107        "help" | "--help" | "-h" => {
108            println!("{USAGE}");
109            ExitCode::SUCCESS
110        }
111        "install" | "deploy" | "uninstall" | "status" | "health" | "logs" | "hooks" => {
112            let parsed = match parse_args(&args[1..]) {
113                Ok(a) => a,
114                Err(e) if e == "help" => {
115                    println!("{USAGE}");
116                    return ExitCode::SUCCESS;
117                }
118                Err(e) => {
119                    eprintln!("{e}\n{USAGE}");
120                    return ExitCode::from(2);
121                }
122            };
123            let (Some(dir), Some(name)) = (parsed.dir, parsed.name) else {
124                eprintln!("tamias-driver-tarsi: --dir and --name are required\n{USAGE}");
125                return ExitCode::from(2);
126            };
127            let inv = drv::Invocation {
128                dir,
129                name,
130                config: parsed.config,
131            };
132            match drv::run_verb(verb, &inv, &env()) {
133                Ok(report) => {
134                    drv::emit(&report);
135                    if report["ok"] == true {
136                        ExitCode::SUCCESS
137                    } else {
138                        ExitCode::FAILURE
139                    }
140                }
141                Err(e) => {
142                    eprintln!("{e}");
143                    ExitCode::FAILURE
144                }
145            }
146        }
147        other => {
148            eprintln!("tamias-driver-tarsi: unknown verb '{other}'\n{USAGE}");
149            ExitCode::from(2)
150        }
151    }
152}

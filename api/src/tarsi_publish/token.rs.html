1//! Publish-token resolution — the seam between config's secret
2//! *names* (`token_env`) / refs (`token_ref`) and the values the
3//! forges consume. Values are secret: they are never logged, never
4//! placed in error messages, and never written anywhere but the
5//! child process environment / `Authorization` header.
6//!
7//! Resolution order (T-041):
8//!
9//! 1. `token_ref` — an explicit `cuora://<scope>/<name>` secret
10//!    ref, fetched through `hypha-secret-cuora` (audit events and
11//!    vault policy apply). Nested scopes work (`cuora://a/b/name`
12//!    → scope `a/b`, name `name` — the ref tail is the secret
13//!    name).
14//! 2. `token_env` — the environment variable of that name.
15//!
16//! A `cuora://` miss falls through to the env var (the org's
17//! Secrets-chain semantics); an unsupported ref scheme is a hard
18//! config error. With no `token_ref`, behavior is exactly the
19//! pre-T-041 env lookup.
20//!
21//! Env knobs follow the `<PRODUCT>_CUORA_*` convention:
22//! `TARSI_CUORA_BIN` overrides the `cuora` binary (tests stub it),
23//! `TARSI_CUORA_TRANSPORT` picks `cli`/`serve`.
24

34/// Env var naming the `cuora` binary override.
35pub const CUORA_BIN_ENV: &str = "TARSI_CUORA_BIN";
36/// Env var picking the cuora transport (`cli`/`serve`).
37pub const CUORA_TRANSPORT_ENV: &str = "TARSI_CUORA_TRANSPORT";
38
39/// Resolves a publish token from config's `(token_env, token_ref)`
40/// pair. Implementations must treat the returned value as secret.
41pub trait TokenResolver: Send + Sync {
42    /// Resolve the token. `token_ref` is an optional
43    /// `cuora://<scope>/<name>` ref taking precedence over the env
44    /// var named `token_env`. `Ok(None)` means no token is
45    /// configured or found — the caller decides whether that's an
46    /// error (publish) or fine (dry-run).
47    ///
48    /// # Errors
49    /// `PublishError::Msg` on an unparseable/unsupported `token_ref`
50    /// or a resolver-side failure. The secret *value* must never
51    /// appear in the error.
52    fn resolve(
53        &self,
54        token_env: &str,
55        token_ref: Option<&str>,
56    ) -> Result<Option<String>, PublishError>;
57}
58
59/// Split a `cuora://<scope>/<name>` ref into `(scope, name)` —
60/// scope may itself contain `/` (e.g. `peregrin/push`), so the
61/// split is at the *last* `/`. Rejects missing/empty parts.
62fn parse_cuora_ref(r: &str) -> Result<(&str, &str), PublishError> {
63    let rest = r
64        .strip_prefix("cuora://")
65        .ok_or_else(|| PublishError::Msg(format!("unsupported token_ref scheme: {r}")))?;
66    let (scope, name) = rest.rsplit_once('/').ok_or_else(|| {
67        PublishError::Msg(format!(
68            "token_ref {r:?} has no secret name (want cuora://<scope>/<name>)"
69        ))
70    })?;
71    if scope.is_empty() || name.is_empty() {
72        return Err(PublishError::Msg(format!(
73            "token_ref {r:?} has an empty scope or name"
74        )));
75    }
76    Ok((scope, name))
77}

79/// The default resolver: `cuora://` refs via `hypha-secret-cuora`,
80/// env var fallback. One [`CuoraSecrets`] store per scope, built
81/// lazily and cached for the resolver's lifetime (cheap under the
82/// `cli` transport; under `serve` it keeps the `cuora serve` child
83/// alive across lookups).
84#[derive(Default)]
85pub struct CuoraTokenResolver {
86    stores: Mutex<HashMap<String, CuoraSecrets>>,
87    /// Test seam — when set, every scope's store uses this `cuora`
88    /// binary (mirrors `CuoraSecrets::with_bin`).
89    bin: Option<PathBuf>,
90}
91
92impl CuoraTokenResolver {
93    /// A resolver honoring `TARSI_CUORA_BIN`/`TARSI_CUORA_TRANSPORT`.
94    #[must_use]
95    pub fn new() -> Self {
96        Self::default()
97    }
98
99    /// Override the `cuora` binary for every scope — the test seam.
100    #[must_use]
101    pub fn with_bin(mut self, bin: PathBuf) -> Self {
102        self.bin = Some(bin);
103        self
104    }
105
106    fn store(&self, scope: &str) -> CuoraSecrets {
107        let store = CuoraSecrets::from_env(scope, CUORA_BIN_ENV, CUORA_TRANSPORT_ENV);
108        match &self.bin {
109            Some(b) => store.with_bin(b.clone()),
110            None => store,
111        }
112    }
113
114    fn cuora_get(&self, scope: &str, name: &str) -> Option<String> {
115        let mut stores = self.stores.lock().ok()?;
116        let store = stores
117            .entry(scope.to_string())
118            .or_insert_with(|| self.store(scope));
119        store.get(name)
120    }
121}
122
123impl TokenResolver for CuoraTokenResolver {
124    fn resolve(
125        &self,
126        token_env: &str,
127        token_ref: Option<&str>,
128    ) -> Result<Option<String>, PublishError> {
129        if let Some(r) = token_ref {
130            let (scope, name) = parse_cuora_ref(r)?;
131            if let Some(v) = self.cuora_get(scope, name) {
132                return Ok(Some(v));
133            }
134            // cuora miss → fall through to the env var.
135        }
136        Ok(std::env::var(token_env).ok().filter(|v| !v.is_empty()))
137    }
138}

140#[cfg(test)]

146    /// A fake `cuora` binary: records its argv to `$CAPTURE` and
147    /// prints `VALUE` on `get`. The capture path is baked into the
148    /// generated script.
149    fn stub(dir: &std::path::Path, capture: &std::path::Path, value: &str) -> PathBuf {
150        let script = format!(
151            "#!/bin/sh\necho \"$@\" > '{}'\nprintf '%s\\n' '{}'\n",
152            capture.display(),
153            value
154        );
155        let bin = dir.join("cuora-stub.sh");
156        std::fs::write(&bin, script).unwrap();
157        std::fs::set_permissions(&bin, std::fs::Permissions::from_mode(0o755)).unwrap();
158        bin
159    }
160
161    #[test]
162    fn parses_nested_scope_refs() {
163        assert_eq!(
164            parse_cuora_ref("cuora://tamias/crates-token").unwrap(),
165            ("tamias", "crates-token")
166        );
167        assert_eq!(
168            parse_cuora_ref("cuora://peregrin/push/vapid").unwrap(),
169            ("peregrin/push", "vapid")
170        );
171        assert!(parse_cuora_ref("https://x/y").is_err());
172        assert!(parse_cuora_ref("cuora://noslash").is_err());
173        assert!(parse_cuora_ref("cuora:///name").is_err());
174        assert!(parse_cuora_ref("cuora://scope/").is_err());
175    }
176
177    #[test]
178    fn token_ref_wins_over_env() {
179        let dir = tempfile::tempdir().unwrap();
180        let capture = dir.path().join("args.txt");
181        let bin = stub(dir.path(), &capture, "ref-secret");
182        std::env::set_var("TARSI_TEST_TOKEN_ENV_ONLY", "env-secret");
183        let r = CuoraTokenResolver::new().with_bin(bin);
184        let v = r
185            .resolve(
186                "TARSI_TEST_TOKEN_ENV_ONLY",
187                Some("cuora://tamias/crates-token"),
188            )
189            .unwrap()
190            .unwrap();
191        assert_eq!(v, "ref-secret");
192        // The CLI leg fetched exactly the configured ref.
193        let args = std::fs::read_to_string(&capture).unwrap();
194        assert!(args.contains("cuora://tamias/crates-token"), "{args}");
195        assert!(args.contains("get"), "{args}");
196    }
197
198    #[test]
199    fn env_is_the_fallback_and_default() {
200        let dir = tempfile::tempdir().unwrap();
201        // A failing cuora: ref misses, env still answers.
202        let bin = dir.path().join("cuora-fail.sh");
203        std::fs::write(&bin, "#!/bin/sh\nexit 1\n").unwrap();
204        std::fs::set_permissions(&bin, std::fs::Permissions::from_mode(0o755)).unwrap();
205        std::env::set_var("TARSI_TEST_TOKEN_FALLBACK", "env-value");
206        let r = CuoraTokenResolver::new().with_bin(bin);
207        // token_ref set but cuora misses → env fallback.
208        let v = r
209            .resolve("TARSI_TEST_TOKEN_FALLBACK", Some("cuora://tamias/missing"))
210            .unwrap()
211            .unwrap();
212        assert_eq!(v, "env-value");
213        // No token_ref → plain env lookup (pre-T-041 behavior).
214        let v = r
215            .resolve("TARSI_TEST_TOKEN_FALLBACK", None)
216            .unwrap()
217            .unwrap();
218        assert_eq!(v, "env-value");
219        // Nothing configured anywhere → None.
220        assert_eq!(r.resolve("TARSI_TEST_TOKEN_UNSET_XYZ", None).unwrap(), None);
221    }

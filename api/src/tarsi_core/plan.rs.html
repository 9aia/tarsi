1//! The release plan — pure computation from unit specs + commits.
2//! `plan_release` attributes commits to units by path prefix,
3//! infers bumps, topo-sorts publish order, and emits the ordered
4//! [`ReleaseStep`]s a driver/serve op would execute.

17/// One unit's resolved release decision.
18#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
19pub struct UnitPlan {
20    pub spec: ReleaseUnitSpec,
21    /// Inferred or overridden bump.
22    pub bump: Bump,
23    pub new_version: Version,
24    pub tag: String,
25    /// Conventional commits attributed to this unit (newest first,
26    /// as supplied).
27    pub commits: Vec<Commit>,
28    /// Parsed views of `commits` (same order).
29    pub conventional: Vec<ConventionalCommit>,
30    /// Rendered changelog block for this unit's release.
31    pub changelog: Changelog,
32}
33
34impl UnitPlan {
35    /// Whether this unit actually releases (bump != none).
36    #[must_use]
37    pub fn releases(&self) -> bool {
38        self.bump.releases()
39    }
40}

42/// An ordered, executable step in the release. Steps are data —
43/// `tarsi-serve`/drivers interpret them; core never performs I/O.
44#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
45#[serde(tag = "op", rename_all = "kebab-case")]
46pub enum ReleaseStep {
47    /// Splice each releasing unit's changelog into its
48    /// `CHANGELOG.md` (or the configured path).
49    WriteChangelogs,
50    /// Bump versions in manifests (`cargo set-version`-equivalent).
51    BumpManifests,
52    /// Commit the version bump.
53    CommitBump,
54    /// Create the unit's tag(s) — one per releasing unit.
55    Tag,
56    /// Publish crates in dependency order (per-unit `cargo publish`).
57    PublishCrates,
58    /// Create GitHub Releases with rendered notes.
59    PublishGithubReleases,
60    /// Push commits + tags.
61    Push,
62}
63
64/// The full plan for one release invocation.
65#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
66pub struct ReleasePlan {
67    /// Units in publish order (topo-sorted on `depends_on`).
68    pub units: Vec<UnitPlan>,
69    /// Commits that matched no unit path — surfaced so callers can
70    /// warn rather than silently drop them.
71    pub unattributed: Vec<Commit>,
72}
73
74impl ReleasePlan {
75    /// Ordered steps this plan implies. Steps appear once per plan
76    /// (not per unit) except publish steps which name their unit.
77    #[must_use]
78    pub fn steps(&self) -> Vec<ReleaseStep> {
79        // Push precedes publish — execution pushes the bump commit
80        // and tags *before* any forge call (GitHub Releases
81        // auto-create `refs/tags/<tag>` at remote HEAD otherwise).
82        let mut steps = vec![
83            ReleaseStep::WriteChangelogs,
84            ReleaseStep::BumpManifests,
85            ReleaseStep::CommitBump,
86            ReleaseStep::Tag,
87            ReleaseStep::Push,
88        ];
89        for u in self.units.iter().filter(|u| u.releases()) {
90            if u.spec
91                .publish
92                .contains(&crate::unit::PublishTarget::CratesIo)
93            {
94                steps.push(ReleaseStep::PublishCrates);
95                break;
96            }
97        }
98        for u in self.units.iter().filter(|u| u.releases()) {
99            if u.spec
100                .publish
101                .contains(&crate::unit::PublishTarget::GithubRelease)
102            {
103                steps.push(ReleaseStep::PublishGithubReleases);
104                break;
105            }
106        }
107        steps
108    }
109
110    /// Units that will actually release.
111    pub fn releasing(&self) -> impl Iterator<Item = &UnitPlan> {
112        self.units.iter().filter(|u| u.releases())
113    }
114}

116#[derive(Debug, thiserror::Error)]
117pub enum PlanError {
118    #[error("unit {0:?} depends on unknown unit {1:?}")]
119    UnknownDep(String, String),
120    #[error("dependency cycle among units: {0}")]
121    Cycle(String),
122    #[error("duplicate unit name {0:?}")]
123    DuplicateUnit(String),
124    #[error("no release units resolved — check units config")]
125    Empty,
126}

128/// Plan a release under the `conventional` strategy — the
129/// pre-P-030 default. Equivalent to
130/// [`plan_release_with_strategy`] with [`Strategy::conventional`].
131///
132/// `specs` may come from [`UnitsSpec::CargoWorkspace`] resolved by
133/// the caller, or be passed directly — this function is pure.
134pub fn plan_release(
135    specs: &[ReleaseUnitSpec],
136    commits: &[Commit],
137    date: Option<String>,
138) -> Result<ReleasePlan, PlanError> {
139    plan_release_with_strategy(specs, commits, date, &Strategy::conventional())
140}
141
142/// Plan a release under an explicit [`Strategy`] (P-030) — bump
143/// inference and changelog sectioning are data-driven; the
144/// `conventional` builtin reproduces the legacy hardcoded policy
145/// byte-identically.
146///
147/// `specs` may come from [`UnitsSpec::CargoWorkspace`] resolved by
148/// the caller, or be passed directly — this function is pure.
149pub fn plan_release_with_strategy(
150    specs: &[ReleaseUnitSpec],
151    commits: &[Commit],
152    date: Option<String>,
153    strategy: &Strategy,
154) -> Result<ReleasePlan, PlanError> {
155    if specs.is_empty() {
156        return Err(PlanError::Empty);
157    }
158    let mut seen = BTreeSet::new();
159    for s in specs {
160        if !seen.insert(s.name.as_str()) {
161            return Err(PlanError::DuplicateUnit(s.name.clone()));
162        }
163        for dep in &s.depends_on {
164            if !specs.iter().any(|o| &o.name == dep) {
165                return Err(PlanError::UnknownDep(s.name.clone(), dep.clone()));
166            }
167        }
168    }
169
170    let mut units = Vec::with_capacity(specs.len());
171    let mut unattributed = Vec::new();
172    for spec in specs {
173        let mine: Vec<Commit> = commits
174            .iter()
175            .filter(|c| commit_touches(c, &spec.path))
176            .cloned()
177            .collect();
178        // Inference fold: a commit contributes the max bump across
179        // its matching rules; the unit takes the max across commits.
180        let mut inferred = Bump::None;
181        let mut conventional = Vec::new();
182        let mut changelog =
183            Changelog::for_strategy(&strategy.changelog.sections, String::new(), date.clone());
184        for c in &mine {
185            let Some(cc) = c.conventional() else {
186                continue;
187            };
188            conventional.push(cc.clone());
189            inferred = inferred.max(strategy.bump_for(&cc));
190            if let Some(sec) = strategy.section_for(&cc) {
191                changelog.push(
192                    &sec.key,
193                    &sec.heading,
194                    ChangelogEntry::from_commit(&c.sha, &cc),
195                );
196            }
197        }
198        // `spec.bump` (unit override) wins outright; otherwise the
199        // inferred bump, falling back to the strategy's `fallback`
200        // (`release.default_bump` is folded in by the caller) when
201        // inference finds nothing.
202        let bump = spec.bump.unwrap_or_else(|| {
203            if inferred.releases() {
204                inferred
205            } else {
206                strategy.bump.fallback.unwrap_or(Bump::None)
207            }
208        });
209        let new_version = bump.apply(&spec.current_version);
210        changelog.version = new_version.to_string();
211        units.push(UnitPlan {
212            tag: spec.tag(&new_version),
213            spec: spec.clone(),
214            bump,
215            new_version,
216            commits: mine,
217            conventional,
218            changelog,
219        });
220    }
221
222    // Unattributed = commits touching no unit path. A commit with
223    // empty `paths` is conservatively attributed to every unit
224    // (unknown granularity), so it never lands here.
225    for c in commits {
226        if !specs.iter().any(|s| commit_touches(c, &s.path)) {
227            unattributed.push(c.clone());
228        }
229    }
230
231    let ordered = topo_sort(&units)?;
232    Ok(ReleasePlan {
233        units: ordered,
234        unattributed,
235    })
236}

238/// Path-prefix attribution. Empty commit `paths` = unknown →
239/// matches every unit (can't prove it doesn't touch the unit).
240/// Empty unit path = whole repo.
241fn commit_touches(commit: &Commit, unit_path: &str) -> bool {
242    if unit_path.is_empty() || unit_path == "." {
243        return true;
244    }
245    if commit.paths.is_empty() {
246        return true;
247    }
248    let prefix = unit_path.trim_end_matches('/');
249    commit
250        .paths
251        .iter()
252        .any(|p| p == prefix || p.starts_with(&format!("{prefix}/")))
253}

255// NOTE: resolving `UnitsSpec::CargoWorkspace` into concrete specs
256// (manifest scanning) is intentionally NOT here — core stays pure.
257// That resolution belongs to a manifest/`tarsi-config` layer that
258// feeds `plan_release` explicit `ReleaseUnitSpec`s.
259
260/// Kahn's algorithm over `depends_on`. Returns units in publish
261/// order — dependencies first.
262fn topo_sort(units: &[UnitPlan]) -> Result<Vec<UnitPlan>, PlanError> {
263    let by_name: BTreeMap<&str, &UnitPlan> =
264        units.iter().map(|u| (u.spec.name.as_str(), u)).collect();
265    let mut indegree: BTreeMap<&str, usize> = BTreeMap::new();
266    let mut dependents: BTreeMap<&str, Vec<&str>> = BTreeMap::new();
267    for u in units {
268        indegree.entry(u.spec.name.as_str()).or_insert(0);
269    }
270    for u in units {
271        for dep in &u.spec.depends_on {
272            dependents
273                .entry(dep.as_str())
274                .or_default()
275                .push(u.spec.name.as_str());
276            if let Some(d) = indegree.get_mut(u.spec.name.as_str()) {
277                *d += 1;
278            }
279        }
280    }
281    // Stable-ish: BTreeMap iteration gives alphabetical tie-break.
282    let mut queue: VecDeque<&str> = indegree
283        .iter()
284        .filter(|(_, &d)| d == 0)
285        .map(|(&n, _)| n)
286        .collect();
287    let mut out = Vec::with_capacity(units.len());
288    while let Some(name) = queue.pop_front() {
289        let unit = by_name[name];
290        out.push(unit.clone());
291        if let Some(deps) = dependents.get(name) {
292            for &dep_name in deps {
293                if let Some(d) = indegree.get_mut(dep_name) {
294                    *d -= 1;
295                    if *d == 0 {
296                        queue.push_back(dep_name);
297                    }
298                }
299            }
300        }
301    }
302    if out.len() != units.len() {
303        let stuck: Vec<&str> = indegree
304            .iter()
305            .filter(|(_, &d)| d > 0)
306            .map(|(&n, _)| n)
307            .collect();
308        return Err(PlanError::Cycle(stuck.join(", ")));
309    }
310    Ok(out)
311}

340    #[test]
341    fn attributes_and_infers_bumps() {
342        let specs = vec![
343            spec("core", "crates/core", &[]),
344            spec("cli", "crates/cli", &["core"]),
345        ];
346        let commits = vec![
347            commit(
348                "aaa1111",
349                "feat(core): add thing",
350                &["crates/core/src/lib.rs"],
351            ),
352            commit("bbb2222", "fix(cli): crash", &["crates/cli/src/main.rs"]),
353            commit(
354                "ccc3333",
355                "feat(cli)!: break api",
356                &["crates/cli/src/lib.rs"],
357            ),
358            commit("ddd4444", "docs: readme", &["README.md"]),
359        ];
360        let plan = plan_release(&specs, &commits, None).unwrap();
361        // topo: core before cli
362        assert_eq!(plan.units[0].spec.name, "core");
363        assert_eq!(plan.units[1].spec.name, "cli");
364        assert_eq!(plan.units[0].bump, Bump::Minor);
365        assert_eq!(plan.units[0].new_version.to_string(), "0.2.0");
366        assert_eq!(plan.units[0].tag, "core-v0.2.0");
367        assert_eq!(plan.units[1].bump, Bump::Major);
368        assert_eq!(plan.units[1].tag, "cli-v1.0.0");
369        // docs commit attributed to both (path "" matches? no —
370        // unit paths are non-empty; README.md touches neither)
371        assert_eq!(plan.unattributed.len(), 1);
372        assert_eq!(plan.unattributed[0].sha, "ddd4444");
373    }

428    #[test]
429    fn steps_follow_publish_targets() {
430        let specs = vec![spec("lib", "", &[])];
431        let commits = vec![commit("x1", "feat: x", &[])];
432        let plan = plan_release(&specs, &commits, None).unwrap();
433        let steps = plan.steps();
434        assert!(matches!(steps[0], ReleaseStep::WriteChangelogs));
435        assert!(steps.contains(&ReleaseStep::PublishCrates));
436        assert!(!steps.contains(&ReleaseStep::PublishGithubReleases));
437        // Push before publish — matches execution order.
438        let push = steps.iter().position(|s| *s == ReleaseStep::Push).unwrap();
439        let publish = steps
440            .iter()
441            .position(|s| *s == ReleaseStep::PublishCrates)
442            .unwrap();
443        assert!(push < publish, "push must precede publish: {steps:?}");
444    }

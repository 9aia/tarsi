1//! Changelog model: strategy-ordered sections, entries grouped
2//! under them, markdown rendering for splicing into
3//! `CHANGELOG.md`. Section identity/order/headings come from the
4//! release [`Strategy`](crate::strategy::Strategy) — P-030 made
5//! the keep-a-changelog section map data, so this module no longer
6//! hardcodes `Section` kinds; the `conventional` builtin
7//! reproduces the old map byte-identically.
8
9use serde::{Deserialize, Serialize};
10
11use crate::commit::ConventionalCommit;
12
13/// One rendered section: stable `key` (the strategy's section
14/// identity), display `heading` (`### ` text), and its entries.
15/// Sections render in `Changelog.sections` order — the strategy's
16/// table order.
17#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
18pub struct ChangelogSection {
19    pub key: String,
20    pub heading: String,
21    pub entries: Vec<ChangelogEntry>,
22}
23
24/// One changelog line's worth of data.
25#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
26pub struct ChangelogEntry {
27    /// Short sha for `(#123)`-style attribution later; empty = omit.
28    pub sha: String,
29    pub scope: Option<String>,
30    pub description: String,
31    pub breaking: bool,
32}
33
34impl ChangelogEntry {
35    /// Build from a parsed conventional commit + its sha.
36    #[must_use]
37    pub fn from_commit(sha: &str, cc: &ConventionalCommit) -> Self {
38        Self {
39            sha: sha.to_string(),
40            scope: cc.scope.clone(),
41            description: cc.description.clone(),
42            breaking: cc.breaking,
43        }
44    }
45}
46
47/// Sections → entries for one release. `sections` is an ordered
48/// `Vec` — order is the strategy's render order (the old
49/// `BTreeMap<Section, _>` relied on the enum's declaration order;
50/// string keys can't carry that, so the strategy's table order is
51/// stored explicitly).
52#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
53pub struct Changelog {
54    /// Rendered under `## [version] - date`; empty version =
55    /// `[Unreleased]`.
56    pub version: String,
57    /// ISO date; `None` = omit the date suffix.
58    pub date: Option<String>,
59    /// Ordered sections — pre-seeded from the strategy's section
60    /// table so render order is stable regardless of which
61    /// section a commit lands in first.
62    pub sections: Vec<ChangelogSection>,
63}
64
65impl Changelog {
66    /// An empty changelog with the strategy's sections pre-seeded
67    /// in render order — every section exists (empty) so `push`
68    /// appends in place and `render` emits table order.
69    #[must_use]
70    pub fn for_strategy(
71        sections: &[crate::strategy::SectionSpec],
72        version: String,
73        date: Option<String>,
74    ) -> Self {
75        Self {
76            version,
77            date,
78            sections: sections
79                .iter()
80                .map(|s| ChangelogSection {
81                    key: s.key.clone(),
82                    heading: s.heading.clone(),
83                    entries: Vec::new(),
84                })
85                .collect(),
86        }
87    }
88
89    /// Append `entry` to section `key`. The section is created at
90    /// the end (with `heading` as its `### ` text) when absent —
91    /// callers that pre-seed via [`Changelog::for_strategy`] never
92    /// hit that path.
93    pub fn push(&mut self, key: &str, heading: &str, entry: ChangelogEntry) {
94        match self.sections.iter_mut().find(|s| s.key == key) {
95            Some(s) => s.entries.push(entry),
96            None => self.sections.push(ChangelogSection {
97                key: key.to_string(),
98                heading: heading.to_string(),
99                entries: vec![entry],
100            }),
101        }
102    }
103
104    /// Render to keep-a-changelog markdown:
105    ///
106    /// ```text
107    /// ## [1.2.0] - 2026-11-10
108    ///
109    /// ### Added
110    ///
111    /// - **scope:** description (`abc1234`)
112    /// ```
113    #[must_use]
114    pub fn render(&self) -> String {
115        let mut out = String::new();
116        let heading = if self.version.is_empty() {
117            "[Unreleased]".to_string()
118        } else {
119            match &self.date {
120                Some(d) => format!("[{}] - {}", self.version, d),
121                None => format!("[{}]", self.version),
122            }
123        };
124        out.push_str("## ");
125        out.push_str(&heading);
126        out.push('\n');
127        for section in &self.sections {
128            if section.entries.is_empty() {
129                continue;
130            }
131            out.push_str("\n### ");
132            out.push_str(&section.heading);
133            out.push_str("\n\n");
134            for e in &section.entries {
135                out.push_str("- ");
136                if e.breaking {
137                    out.push_str("**BREAKING** ");
138                }
139                if let Some(scope) = &e.scope {
140                    out.push_str(&format!("**{scope}:** "));
141                }
142                out.push_str(&e.description);
143                if !e.sha.is_empty() {
144                    out.push_str(&format!(" (`{}`)", short_sha(&e.sha)));
145                }
146                out.push('\n');
147            }
148        }
149        out
150    }
151}
152
153fn short_sha(sha: &str) -> &str {
154    sha.get(..7).unwrap_or(sha)
155}

157#[cfg(test)]
158#[allow(clippy::unwrap_used)]
159mod tests {
160    use super::*;
161    use crate::commit::parse_conventional;
162    use crate::strategy::Strategy;
163
164    #[test]
165    fn sections_render_in_strategy_order() {
166        let strategy = Strategy::conventional();
167        let mut cl = Changelog::for_strategy(
168            &strategy.changelog.sections,
169            "1.0.0".into(),
170            Some("2026-11-10".into()),
171        );
172        // Push Fixed before Added — render order must still be
173        // the strategy's table order (Added before Fixed… via
174        // Changed between them; only non-empty sections render).
175        for subject in ["fix: crash on empty input", "feat: new thing"] {
176            let cc = parse_conventional(subject).unwrap();
177            let sec = strategy.section_for(&cc).unwrap();
178            cl.push(
179                &sec.key,
180                &sec.heading,
181                ChangelogEntry::from_commit("abcdef1234567890", &cc),
182            );
183        }
184        let out = cl.render();
185        let added = out.find("### Added").unwrap();
186        let fixed = out.find("### Fixed").unwrap();
187        assert!(added < fixed, "Added must render before Fixed");
188        assert!(out.contains("## [1.0.0] - 2026-11-10"));
189        assert!(out.contains("new thing (`abcdef1`)"));
190    }
191
192    #[test]
193    fn empty_sections_are_skipped_and_unknown_keys_append() {
194        let mut cl = Changelog::default();
195        cl.push(
196            "custom",
197            "Custom Heading",
198            ChangelogEntry {
199                sha: "x".into(),
200                scope: None,
201                description: "thing".into(),
202                breaking: false,
203            },
204        );
205        let out = cl.render();
206        assert!(out.contains("### Custom Heading"));
207        assert!(out.contains("- thing"));
208        // Empty sections never render headings.
209        assert!(!out.contains("### Added"));
210    }

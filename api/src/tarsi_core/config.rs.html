1//! The `tarsi.toml`/`tarsi.yml` config model — pure serde types.
2//! Loading/discovery lives in scarab-config (`FileSource`); this
3//! module only defines what the file *means*. Secret **names**
4//! and `cuora://` refs appear here (`token_env`, `token_ref`);
5//! values never do — cuora boundary.

15/// Top-level config — `tarsi.toml` or `tarsi.yaml`.
16#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
17pub struct Config {
18    /// Named release profile to layer *under* this config — a
19    /// `type: tarsi` tamias pack or a dir under
20    /// `~/.config/tarsi/profiles/`. Resolved by `tarsi-config`
21    /// (profile dir becomes a config layer between the repo file
22    /// and the user file); ignored by planning itself.
23    #[serde(default, skip_serializing_if = "Option::is_none")]
24    pub profile: Option<String>,
25    #[serde(default)]
26    pub release: ReleaseConfig,
27    #[serde(default)]
28    pub publish: PublishConfig,
29}

31#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
32pub struct ReleaseConfig {
33    /// How release units resolve. Default: cargo workspace at root.
34    #[serde(default)]
35    pub units: Option<UnitsSpec>,
36    /// Default tag template for units that don't override it.
37    #[serde(default = "default_tag_format")]
38    pub tag_format: String,
39    /// Changelog file spliced per unit — resolved relative to each
40    /// unit's path (`crates/foo` + `CHANGES.md` →
41    /// `crates/foo/CHANGES.md`).
42    #[serde(default = "default_changelog_path")]
43    pub changelog_path: String,
44    /// keep-a-changelog | conventional-commit-grouped | …
45    ///
46    /// **Deprecated (P-030)** — superseded by
47    /// [`ReleaseConfig::strategy`]'s `changelog` section table.
48    /// Still parsed for backward compat; has no effect on
49    /// planning (it never did — the field was always descriptive).
50    #[serde(default = "default_styleguide")]
51    pub styleguide: String,
52    /// Release strategy (P-030): a builtin name
53    /// (`"conventional"`, `"semantic-release-angular"`) or an
54    /// inline rules table (`[release.strategy.bump]` /
55    /// `[release.strategy.changelog]`). Absent = `conventional`,
56    /// the pre-strategy behavior. Resolved inside `tarsi-serve`'s
57    /// plan path so every op inherits it.
58    #[serde(default, skip_serializing_if = "Option::is_none")]
59    pub strategy: Option<StrategyRef>,
60    /// Repo-wide default bump when inference finds nothing —
61    /// honored as the strategy's `fallback` (P-030 wired the
62    /// previously-dead field; an explicit `strategy.bump.fallback`
63    /// wins).
64    #[serde(default)]
65    pub default_bump: Option<Bump>,
66    /// Path globs excluded from commit→unit attribution
67    /// (e.g. `["docs/**", "*.md"]`). Wired in P-030: matching
68    /// paths are stripped from each commit before attribution;
69    /// a commit left with no paths is dropped entirely.
70    #[serde(default)]
71    pub diff_exclude: Vec<String>,
72    /// Per-unit overrides keyed by unit name.
73    #[serde(default)]
74    pub unit_overrides: BTreeMap<String, UnitOverride>,
75    /// Pack-ref/doc version sync — rewrite stale `name@old` /
76    /// manifest `version:` refs in repo files *before* the release
77    /// commit lands, so the tagged tree is self-consistent (T-039;
78    /// replaces the per-repo `scripts/release.sh` pre-step). Empty
79    /// `rules` = disabled — repos opt in.
80    #[serde(default)]
81    pub sync_refs: SyncRefsConfig,
82}
83
84impl Default for ReleaseConfig {
85    fn default() -> Self {
86        Self {
87            units: None,
88            tag_format: default_tag_format(),
89            changelog_path: default_changelog_path(),
90            styleguide: default_styleguide(),
91            strategy: None,
92            default_bump: None,
93            diff_exclude: Vec::new(),
94            unit_overrides: BTreeMap::new(),
95            sync_refs: SyncRefsConfig::default(),
96        }
97    }
98}

110/// `[release.sync_refs]` — the pack-ref sync stage config. Runs
111/// inside `bump`/`release` after manifests are bumped and before
112/// the release commit, so synced refs ride the tagged tree.
113#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
114pub struct SyncRefsConfig {
115    /// Rewrite rules, applied in order. Empty = the stage is
116    /// disabled entirely (no files touched, no step reported).
117    #[serde(default)]
118    pub rules: Vec<SyncRefRule>,
119}
120
121/// One sync rewrite: every regex match of `find` in each matched
122/// file is replaced by the literal `replace`.
123///
124/// `{old}`/`{new}` placeholders expand to the releasing units'
125/// shared current/new versions — `{old}`/`{new}` inside `find`
126/// are regex-escaped before compilation (so `.` stays literal),
127/// inside `replace` they expand literally. Patterns are not
128/// auto-bounded — `app@{old}` still matches the `app@1.2.3`
129/// prefix of `app@1.2.30`, exactly like the old `sed` pre-steps;
130/// add `\b` (or similar) when a boundary is needed. Line anchors
131/// need the regex multi-line flag: `(?m)^version: "{old}"`.
132#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
133pub struct SyncRefRule {
134    /// Repo-root-relative globs of files to rewrite
135    /// (`"libraries/myco/tamias.yaml"`, `"README.md"`,
136    /// `".agents/skills/myco/SKILL.md"`). Non-matching globs are
137    /// skipped — a repo can carry rules for files it doesn't have.
138    #[serde(default)]
139    pub files: Vec<String>,
140    /// Regex matching the stale ref. Required.
141    pub find: String,
142    /// Literal replacement text. Required — an empty string would
143    /// silently delete refs, so serde rejects a missing key.
144    pub replace: String,
145}

179#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
180pub struct CratesPublish {
181    /// crates.io publishing on/off.
182    #[serde(default = "default_true")]
183    pub enabled: bool,
184    /// Retry crates.io 429s with backoff (idempotent re-runs).
185    #[serde(default = "default_true")]
186    pub retry_429: bool,
187    /// Max publish attempts per unit.
188    #[serde(default = "default_attempts")]
189    pub max_attempts: u32,
190    /// Per-attempt wall-clock cap for `cargo publish` in seconds
191    /// (verify build + upload). Heavy crates need minutes — the
192    /// default covers tamias-serve/tamias-tui-class builds
193    /// (I-003). `0` disables the cap.
194    #[serde(default = "default_timeout_secs")]
195    pub timeout_secs: u64,
196    /// Base delay in seconds between 429 retries — doubles per
197    /// attempt up to `retry_max_secs`. crates.io bulk throttles
198    /// ask for minutes-scale spacing (I-003).
199    #[serde(default = "default_retry_base_secs")]
200    pub retry_base_secs: u64,
201    /// Ceiling in seconds for computed 429 backoff. A server
202    /// `Retry-After` hint is honored verbatim and may exceed it.
203    #[serde(default = "default_retry_max_secs")]
204    pub retry_max_secs: u64,
205    /// Env var NAME consulted for the crates.io token when
206    /// `token_ref` is unset or misses — cargo reads
207    /// `CARGO_REGISTRY_TOKEN`, and a resolved value is injected
208    /// into the publish child's environment under that name. The
209    /// value never appears in config, manifests, or logs.
210    #[serde(default = "default_crates_token_env")]
211    pub token_env: String,
212    /// Optional `cuora://<scope>/<name>` secret ref — resolved via
213    /// cuora at publish time and takes precedence over `token_env`.
214    /// The value never appears in config or logs.
215    #[serde(default, skip_serializing_if = "Option::is_none")]
216    pub token_ref: Option<String>,
217}

247#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
248pub struct GhReleasePublish {
249    /// GitHub Releases on/off.
250    #[serde(default = "default_true")]
251    pub enabled: bool,
252    /// Env var NAME holding the GitHub token — resolved via
253    /// environment/cuora at publish time; the value never appears
254    /// in config, manifests, or logs.
255    #[serde(default = "default_token_env")]
256    pub token_env: String,
257    /// Repo override `owner/name`; default = git remote.
258    #[serde(default)]
259    pub repo: Option<String>,
260    /// Optional `cuora://<scope>/<name>` secret ref — resolved via
261    /// cuora at publish time and takes precedence over `token_env`.
262    /// The value never appears in config or logs.
263    #[serde(default, skip_serializing_if = "Option::is_none")]
264    pub token_ref: Option<String>,
265    /// Repo-root-relative glob patterns for release assets —
266    /// `{unit}` and `{version}` expand per releasing unit, e.g.
267    /// `["dist/{unit}-{version}-*", "{unit}/checksums.txt"]`.
268    /// A pattern with no glob magic must match an existing file;
269    /// magic patterns matching nothing are skipped. Empty =
270    /// notes-only releases.
271    #[serde(default)]
272    pub assets: Vec<String>,
273}

297/// Merge config `unit_overrides` onto resolved specs — produces
298/// the final specs `plan_release` consumes. Pure; manifest
299/// resolution stays with the caller.
300#[must_use]
301pub fn apply_overrides(mut specs: Vec<ReleaseUnitSpec>, cfg: &Config) -> Vec<ReleaseUnitSpec> {
302    specs.retain(|s| {
303        !cfg.release
304            .unit_overrides
305            .get(&s.name)
306            .is_some_and(|o| o.skip)
307    });
308    for s in &mut specs {
309        // Repo/profile-level default tag format: specs still
310        // carrying the built-in default (or none) adopt
311        // `release.tag_format`. Per-unit `unit_overrides` below
312        // always win, as does any spec format that differs from
313        // the built-in default.
314        if s.tag_format.is_empty() || s.tag_format == crate::unit::DEFAULT_TAG_FORMAT {
315            s.tag_format = cfg.release.tag_format.clone();
316        }
317        if let Some(o) = cfg.release.unit_overrides.get(&s.name) {
318            if o.bump.is_some() {
319                s.bump = o.bump;
320            }
321            if let Some(t) = &o.tag_format {
322                s.tag_format = t.clone();
323            }
324            if let Some(p) = &o.publish {
325                s.publish = p.clone();
326            }
327        }
328    }
329    specs
330}

337    #[test]
338    fn toml_round_trip() {
339        let text = r#"
340[release]
341changelog_path = "CHANGES.md"
342default_bump = "patch"
343
344[release.unit_overrides."tamias-core"]
345skip = true
346
347[publish.ghreleases]
348token_env = "GH_TOKEN"
349assets = ["dist/{unit}-{version}-*"]
350"#;
351        let cfg: Config = toml::from_str(text).unwrap();
352        assert_eq!(cfg.release.changelog_path, "CHANGES.md");
353        assert_eq!(cfg.release.default_bump, Some(Bump::Patch));
354        assert!(cfg
355            .release
356            .unit_overrides
357            .get("tamias-core")
358            .is_some_and(|o| o.skip));
359        assert_eq!(cfg.publish.ghreleases.token_env, "GH_TOKEN");
360        assert_eq!(
361            cfg.publish.ghreleases.assets,
362            vec!["dist/{unit}-{version}-*".to_string()]
363        );
364        assert!(cfg.publish.crates.enabled);
365    }

367    #[test]
368    fn publish_token_fields() {
369        let text = r#"
370[publish.crates]
371token_env = "CARGO_REGISTRY_TOKEN"
372token_ref = "cuora://tamias/crates-token"
373
374[publish.ghreleases]
375token_ref = "cuora://org/GH_RELEASES_TOKEN"
376"#;
377        let cfg: Config = toml::from_str(text).unwrap();
378        assert_eq!(cfg.publish.crates.token_env, "CARGO_REGISTRY_TOKEN");
379        assert_eq!(
380            cfg.publish.crates.token_ref.as_deref(),
381            Some("cuora://tamias/crates-token")
382        );
383        assert_eq!(
384            cfg.publish.ghreleases.token_ref.as_deref(),
385            Some("cuora://org/GH_RELEASES_TOKEN")
386        );
387        // Defaults: crates reads CARGO_REGISTRY_TOKEN, ghreleases
388        // GITHUB_TOKEN; no refs unless configured.
389        let d = Config::default();
390        assert_eq!(d.publish.crates.token_env, "CARGO_REGISTRY_TOKEN");
391        assert_eq!(d.publish.ghreleases.token_env, "GITHUB_TOKEN");
392        assert!(d.publish.crates.token_ref.is_none());
393        assert!(d.publish.ghreleases.token_ref.is_none());
394    }

396    #[test]
397    fn sync_refs_round_trip() {
398        let text = r#"
399[release.sync_refs]
400rules = [
401  { files = ["libraries/app/tamias.yaml"], find = "(?m)^version: \"{old}\"", replace = "version: \"{new}\"" },
402  { files = ["README.md", ".agents/skills/app/SKILL.md"], find = "app@{old}", replace = "app@{new}" },
403]
404"#;
405        let cfg: Config = toml::from_str(text).unwrap();
406        let rules = &cfg.release.sync_refs.rules;
407        assert_eq!(rules.len(), 2);
408        assert_eq!(rules[0].files, vec!["libraries/app/tamias.yaml"]);
409        assert_eq!(rules[1].replace, "app@{new}");
410        // Missing find/replace is a hard config error.
411        assert!(
412            toml::from_str::<Config>("[release.sync_refs]\nrules = [{ files = [\"x\"] }]").is_err()
413        );
414        // Empty rules = disabled.
415        assert!(Config::default().release.sync_refs.rules.is_empty());
416    }

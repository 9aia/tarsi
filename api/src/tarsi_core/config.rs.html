1//! The `tarsi.toml`/`tarsi.yml` config model — pure serde types.
2//! Loading/discovery lives in scarab-config (`FileSource`); this
3//! module only defines what the file *means*. Secret **names**
4//! and `cuora://` refs appear here (`token_env`, `token_ref`);
5//! values never do — cuora boundary.
6

15/// Top-level config — `tarsi.toml` or `tarsi.yaml`.
16#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
17pub struct Config {
18    /// Named release profile to layer *under* this config — a
19    /// `type: tarsi` tamias pack or a dir under
20    /// `~/.config/tarsi/profiles/`. Resolved by `tarsi-config`
21    /// (profile dir becomes a config layer between the repo file
22    /// and the user file); ignored by planning itself.
23    #[serde(default, skip_serializing_if = "Option::is_none")]
24    pub profile: Option<String>,
25    #[serde(default)]
26    pub release: ReleaseConfig,
27    #[serde(default)]
28    pub publish: PublishConfig,
29}

31#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
32pub struct ReleaseConfig {
33    /// How release units resolve. Default: cargo workspace at root.
34    #[serde(default)]
35    pub units: Option<UnitsSpec>,
36    /// Default tag template for units that don't override it.
37    #[serde(default = "default_tag_format")]
38    pub tag_format: String,
39    /// Changelog file spliced per unit — resolved relative to each
40    /// unit's path (`crates/foo` + `CHANGES.md` →
41    /// `crates/foo/CHANGES.md`).
42    #[serde(default = "default_changelog_path")]
43    pub changelog_path: String,
44    /// keep-a-changelog | conventional-commit-grouped | …
45    ///
46    /// **Deprecated (P-030)** — superseded by
47    /// [`ReleaseConfig::strategy`]'s `changelog` section table.
48    /// Still parsed for backward compat; has no effect on
49    /// planning (it never did — the field was always descriptive).
50    #[serde(default = "default_styleguide")]
51    pub styleguide: String,
52    /// Release strategy (P-030): a builtin name
53    /// (`"conventional"`, `"semantic-release-angular"`) or an
54    /// inline rules table (`[release.strategy.bump]` /
55    /// `[release.strategy.changelog]`). Absent = `conventional`,
56    /// the pre-strategy behavior. Resolved inside `tarsi-serve`'s
57    /// plan path so every op inherits it.
58    #[serde(default, skip_serializing_if = "Option::is_none")]
59    pub strategy: Option<StrategyRef>,
60    /// Repo-wide default bump when inference finds nothing —
61    /// honored as the strategy's `fallback` (P-030 wired the
62    /// previously-dead field; an explicit `strategy.bump.fallback`
63    /// wins).
64    #[serde(default)]
65    pub default_bump: Option<Bump>,
66    /// Path globs excluded from commit→unit attribution
67    /// (e.g. `["docs/**", "*.md"]`). Wired in P-030: matching
68    /// paths are stripped from each commit before attribution;
69    /// a commit left with no paths is dropped entirely.
70    #[serde(default)]
71    pub diff_exclude: Vec<String>,
72    /// Per-unit overrides keyed by unit name.
73    #[serde(default)]
74    pub unit_overrides: BTreeMap<String, UnitOverride>,
75    /// Pack-ref/doc version sync — rewrite stale `name@old` /
76    /// manifest `version:` refs in repo files *before* the release
77    /// commit lands, so the tagged tree is self-consistent (T-039;
78    /// replaces the per-repo `scripts/release.sh` pre-step). Empty
79    /// `rules` = disabled — repos opt in.
80    #[serde(default)]
81    pub sync_refs: SyncRefsConfig,
82}
83
84impl Default for ReleaseConfig {
85    fn default() -> Self {
86        Self {
87            units: None,
88            tag_format: default_tag_format(),
89            changelog_path: default_changelog_path(),
90            styleguide: default_styleguide(),
91            strategy: None,
92            default_bump: None,
93            diff_exclude: Vec::new(),
94            unit_overrides: BTreeMap::new(),
95            sync_refs: SyncRefsConfig::default(),
96        }
97    }
98}

110/// `[release.sync_refs]` — the pack-ref sync stage config. Runs
111/// inside `bump`/`release` after manifests are bumped and before
112/// the release commit, so synced refs ride the tagged tree.
113#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
114pub struct SyncRefsConfig {
115    /// Rewrite rules, applied in order. Empty = the stage is
116    /// disabled entirely (no files touched, no step reported).
117    #[serde(default)]
118    pub rules: Vec<SyncRefRule>,
119}
120
121/// One sync rewrite: every regex match of `find` in each matched
122/// file is replaced by the literal `replace`.
123///
124/// `{old}`/`{new}` placeholders expand to the releasing units'
125/// shared current/new versions — `{old}`/`{new}` inside `find`
126/// are regex-escaped before compilation (so `.` stays literal),
127/// inside `replace` they expand literally. Patterns are not
128/// auto-bounded — `app@{old}` still matches the `app@1.2.3`
129/// prefix of `app@1.2.30`, exactly like the old `sed` pre-steps;
130/// add `\b` (or similar) when a boundary is needed. Line anchors
131/// need the regex multi-line flag: `(?m)^version: "{old}"`.
132#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
133pub struct SyncRefRule {
134    /// Repo-root-relative globs of files to rewrite
135    /// (`"libraries/myco/tamias.yaml"`, `"README.md"`,
136    /// `".agents/skills/myco/SKILL.md"`). Non-matching globs are
137    /// skipped — a repo can carry rules for files it doesn't have.
138    #[serde(default)]
139    pub files: Vec<String>,
140    /// Regex matching the stale ref. Required.
141    pub find: String,
142    /// Literal replacement text. Required — an empty string would
143    /// silently delete refs, so serde rejects a missing key.
144    pub replace: String,
145}

172#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
173pub struct CratesPublish {
174    #[serde(default = "default_true")]
175    pub enabled: bool,
176    /// Retry crates.io 429s with backoff (release.sh semantics).
177    #[serde(default = "default_true")]
178    pub retry_429: bool,
179    /// Max publish attempts per unit.
180    #[serde(default = "default_attempts")]
181    pub max_attempts: u32,
182    /// Env var NAME consulted for the crates.io token when
183    /// `token_ref` is unset or misses — cargo reads
184    /// `CARGO_REGISTRY_TOKEN`, and a resolved value is injected
185    /// into the publish child's environment under that name. The
186    /// value never appears in config, manifests, or logs.
187    #[serde(default = "default_crates_token_env")]
188    pub token_env: String,
189    /// Optional `cuora://<scope>/<name>` secret ref — resolved via
190    /// cuora at publish time and takes precedence over `token_env`.
191    /// The value never appears in config or logs.
192    #[serde(default, skip_serializing_if = "Option::is_none")]
193    pub token_ref: Option<String>,
194}

212#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
213pub struct GhReleasePublish {
214    #[serde(default = "default_true")]
215    pub enabled: bool,
216    /// Env var NAME holding the GitHub token — resolved via
217    /// environment/cuora at publish time; the value never appears
218    /// in config, manifests, or logs.
219    #[serde(default = "default_token_env")]
220    pub token_env: String,
221    /// Repo override `owner/name`; default = git remote.
222    #[serde(default)]
223    pub repo: Option<String>,
224    /// Optional `cuora://<scope>/<name>` secret ref — resolved via
225    /// cuora at publish time and takes precedence over `token_env`.
226    /// The value never appears in config or logs.
227    #[serde(default, skip_serializing_if = "Option::is_none")]
228    pub token_ref: Option<String>,
229    /// Repo-root-relative glob patterns for release assets —
230    /// `{unit}` and `{version}` expand per releasing unit, e.g.
231    /// `["dist/{unit}-{version}-*", "{unit}/checksums.txt"]`.
232    /// A pattern with no glob magic must match an existing file;
233    /// magic patterns matching nothing are skipped. Empty =
234    /// notes-only releases.
235    #[serde(default)]
236    pub assets: Vec<String>,
237}

261/// Merge config `unit_overrides` onto resolved specs — produces
262/// the final specs `plan_release` consumes. Pure; manifest
263/// resolution stays with the caller.
264#[must_use]
265pub fn apply_overrides(mut specs: Vec<ReleaseUnitSpec>, cfg: &Config) -> Vec<ReleaseUnitSpec> {
266    specs.retain(|s| {
267        !cfg.release
268            .unit_overrides
269            .get(&s.name)
270            .is_some_and(|o| o.skip)
271    });
272    for s in &mut specs {
273        // Repo/profile-level default tag format: specs still
274        // carrying the built-in default (or none) adopt
275        // `release.tag_format`. Per-unit `unit_overrides` below
276        // always win, as does any spec format that differs from
277        // the built-in default.
278        if s.tag_format.is_empty() || s.tag_format == crate::unit::DEFAULT_TAG_FORMAT {
279            s.tag_format = cfg.release.tag_format.clone();
280        }
281        if let Some(o) = cfg.release.unit_overrides.get(&s.name) {
282            if o.bump.is_some() {
283                s.bump = o.bump;
284            }
285            if let Some(t) = &o.tag_format {
286                s.tag_format = t.clone();
287            }
288            if let Some(p) = &o.publish {
289                s.publish = p.clone();
290            }
291        }
292    }
293    specs
294}

301    #[test]
302    fn toml_round_trip() {
303        let text = r#"
304[release]
305changelog_path = "CHANGES.md"
306default_bump = "patch"
307
308[release.unit_overrides."tamias-core"]
309skip = true
310
311[publish.ghreleases]
312token_env = "GH_TOKEN"
313assets = ["dist/{unit}-{version}-*"]
314"#;
315        let cfg: Config = toml::from_str(text).unwrap();
316        assert_eq!(cfg.release.changelog_path, "CHANGES.md");
317        assert_eq!(cfg.release.default_bump, Some(Bump::Patch));
318        assert!(cfg
319            .release
320            .unit_overrides
321            .get("tamias-core")
322            .is_some_and(|o| o.skip));
323        assert_eq!(cfg.publish.ghreleases.token_env, "GH_TOKEN");
324        assert_eq!(
325            cfg.publish.ghreleases.assets,
326            vec!["dist/{unit}-{version}-*".to_string()]
327        );
328        assert!(cfg.publish.crates.enabled);
329    }

331    #[test]
332    fn publish_token_fields() {
333        let text = r#"
334[publish.crates]
335token_env = "CARGO_REGISTRY_TOKEN"
336token_ref = "cuora://tamias/crates-token"
337
338[publish.ghreleases]
339token_ref = "cuora://org/GH_RELEASES_TOKEN"
340"#;
341        let cfg: Config = toml::from_str(text).unwrap();
342        assert_eq!(cfg.publish.crates.token_env, "CARGO_REGISTRY_TOKEN");
343        assert_eq!(
344            cfg.publish.crates.token_ref.as_deref(),
345            Some("cuora://tamias/crates-token")
346        );
347        assert_eq!(
348            cfg.publish.ghreleases.token_ref.as_deref(),
349            Some("cuora://org/GH_RELEASES_TOKEN")
350        );
351        // Defaults: crates reads CARGO_REGISTRY_TOKEN, ghreleases
352        // GITHUB_TOKEN; no refs unless configured.
353        let d = Config::default();
354        assert_eq!(d.publish.crates.token_env, "CARGO_REGISTRY_TOKEN");
355        assert_eq!(d.publish.ghreleases.token_env, "GITHUB_TOKEN");
356        assert!(d.publish.crates.token_ref.is_none());
357        assert!(d.publish.ghreleases.token_ref.is_none());
358    }

360    #[test]
361    fn sync_refs_round_trip() {
362        let text = r#"
363[release.sync_refs]
364rules = [
365  { files = ["libraries/app/tamias.yaml"], find = "(?m)^version: \"{old}\"", replace = "version: \"{new}\"" },
366  { files = ["README.md", ".agents/skills/app/SKILL.md"], find = "app@{old}", replace = "app@{new}" },
367]
368"#;
369        let cfg: Config = toml::from_str(text).unwrap();
370        let rules = &cfg.release.sync_refs.rules;
371        assert_eq!(rules.len(), 2);
372        assert_eq!(rules[0].files, vec!["libraries/app/tamias.yaml"]);
373        assert_eq!(rules[1].replace, "app@{new}");
374        // Missing find/replace is a hard config error.
375        assert!(
376            toml::from_str::<Config>("[release.sync_refs]\nrules = [{ files = [\"x\"] }]").is_err()
377        );
378        // Empty rules = disabled.
379        assert!(Config::default().release.sync_refs.rules.is_empty());
380    }
